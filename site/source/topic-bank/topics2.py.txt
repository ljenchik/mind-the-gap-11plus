def lcm(a, b): return a * b // gcd(a, b)


# ---------- 11 Rounding ----------
def rnd_near(r):
    k = r.choice([10, 100, 1000]); N = r.randint(k * 2 + 1, k * 95)
    while N % k == 0: N += 1
    ctx = r.choice(['Write {} to the nearest {}.', 'A river is {} km long. Write its length to the nearest {} km.', 'A mountain is {} feet high. Write its height to the nearest {} feet.',
                    '{} people watched a match. Round this to the nearest {}.', 'Round {} to the nearest {}.'])
    a = (N + k // 2) // k * k
    return Q(ctx.format(n(N), n(k)), n(a), f'{n(N)} lies between {n(N // k * k)} and {n(N // k * k + k)}. Halfway is {n(N // k * k + k // 2)}, so it rounds {"up" if a > N else "down"} to {n(a)}.', 'Nearest 10, 100, 1000')


def rnd_whole(r):
    x = D(r.randint(1, 4999)) / r.choice([100, 100, 1000])
    while x % 1 == 0: x += D('0.03')
    a = int(x + D('0.5'))
    return Q(f'Write {dec(x)} to the nearest whole number.', a, f'The tenths digit is {int(x * 10) % 10}, so round {"up" if a > x else "down"}: {a}.', 'Nearest whole number')


def rnd_pound(r):
    p = r.randint(101, 9999)
    while p % 100 == 0: p += 7
    a = (p + 50) // 100
    return Q(f'Write {money(p)} correct to the nearest £1.', f'£{a}', f'{p % 100}p is {"50p or more" if p % 100 >= 50 else "less than 50p"}, so round {"up" if p % 100 >= 50 else "down"} to £{a}.', 'Nearest pound')


def rnd_big(r):
    k = r.choice([10_000, 100_000, 1_000_000]); N = r.randint(11_000_000, 180_000_000) // 1000 * 1000
    while N % k == 0: N += 7000
    a = (N + k // 2) // k * k; th = r.choice(['An ocean covers {} square km.', 'A country has a population of {}.', 'A company sold {} tickets last year.', 'A desert covers {} square miles.'])
    return Q(th.format(n(N)) + f' Round this number to the nearest {n(k)}.', n(a), f'Look at the digit to the right of the {n(k)}s place. {n(N)} rounds {"up" if a > N else "down"} to {n(a)}.', 'Rounding large numbers')


def rnd_dp(r):
    dp = r.choice([1, 2]); x = D(r.randint(1001, 99999)) / 1000
    while (x * 10 ** dp * 10) % 10 == 0: x += D('0.003')
    q = D(1).scaleb(-dp); a = (x / q + D('0.5')).to_integral_value(rounding='ROUND_FLOOR') * q
    return Q(f'Round {dec(x)} to {dp} decimal place{"s" if dp > 1 else ""}.', f'{a:.{dp}f}', f'Look at the next digit after the {["tenths", "hundredths"][dp - 1]}: it is {int(x * 10 ** (dp + 1)) % 10}, so round {"up" if a > x else "down"} to {a:.{dp}f}.', 'Decimal places')


def rnd_estimate(r):
    k = r.randint(6, 19); rp = r.randint(3, 19); off = r.choice([-5, -2, -1, 1, 2, 5, -9, 9]); it = r.choice(['films', 'books', 'T-shirts', 'tickets', 'games', 'plants'])
    return Q(f'By rounding the price to the nearest pound, find the approximate cost of {k} {it} at {money(rp * 100 + off)} each.', f'£{k * rp}',
             f'{money(rp * 100 + off)} is about £{rp}. {k} × £{rp} = £{k * rp}.', 'Estimating')


def rnd_bounds(r):
    k = r.choice([10, 100, 1000]); X = k * r.randint(3, 95); big = r.random() < .5
    a = X + k // 2 - 1 if big else X - k // 2
    return Q(f'A whole number is rounded to the nearest {n(k)} and the answer is {n(X)}. What is the {"largest" if big else "smallest"} number it could have been?', n(a),
             f'Numbers from {n(X - k // 2)} up to {n(X + k // 2 - 1)} round to {n(X)}. ({n(X + k // 2)} would round up to {n(X + k)}.)', 'Largest and smallest possible')

# ---------- 13 Factors, multiples and primes ----------
def fpm_factors(r):
    N = r.choice([x for x in range(12, 121) if 4 <= len(factors(x)) <= 12])
    return Q(f'Write down all the factors of {N}.', ', '.join(map(str, factors(N))), 'Work in pairs: ' + ', '.join(f'{d} × {N // d}' for d in factors(N) if d * d <= N) + '.', 'Factors')


def fpm_common(r):
    while True:
        a, b = r.sample(range(8, 73), 2); c = [d for d in factors(a) if b % d == 0]
        if 3 <= len(c) <= 6 and a % b and b % a: break
    return Q(f'Which numbers are factors of both {a} and {b}?', ', '.join(map(str, c)), f'Factors of {a}: {", ".join(map(str, factors(a)))}. Factors of {b}: {", ".join(map(str, factors(b)))}. The numbers in both lists are {", ".join(map(str, c))}.', 'Common factors')


def fpm_hcf(r):
    g = r.randint(4, 18)
    while True:
        a, b = r.sample(range(2, 10), 2)
        if gcd(a, b) == 1: break
    return Q(f'What is the highest common factor of {a * g} and {b * g}?', g, f'{a * g} = {g} × {a} and {b * g} = {g} × {b}; {a} and {b} share no other factor, so the HCF is {g}.', 'Highest common factor')


def fpm_lcm(r):
    while True:
        a, b = r.sample(range(4, 21), 2)
        if a % b and b % a and lcm(a, b) <= 180: break
    L = lcm(a, b)
    return Q(f'What is the lowest common multiple of {a} and {b}?', L, f'Multiples of {max(a, b)}: {", ".join(str(max(a, b) * i) for i in range(1, L // max(a, b) + 1))}. The first one that {min(a, b)} divides into is {L}.', 'Lowest common multiple')

        a, b, c = r.sample([4, 5, 6, 8, 9, 10, 12, 15, 16, 18, 20, 24, 30], 3); L = lcm(lcm(a, b), c)
        if 30 <= L <= 240 and L not in (a, b, c): break
    st = 60 * r.randint(6, 10) + r.choice([0, 0, 15, 30]); th = r.choice(['buses leave the bus station', 'lighthouses flash together', 'bells ring together', 'trams leave the depot'])
    return Q(f'Three {th} at {t12(st)}. After that, the first repeats every {a} minutes, the second every {b} minutes and the third every {c} minutes. At what time will all three next happen together?', t12(st + L),
             f'The lowest common multiple of {a}, {b} and {c} is {L} minutes ({dur(L)}). {t12(st)} + {dur(L)} = {t12(st + L)}.', 'Common multiples in time')

def pow_between(r):
    lo = r.randint(5, 150); hi = lo + r.randint(40, 110); cube = r.random() < .3
    xs = [i ** 3 for i in range(1, 12) if lo < i ** 3 < hi] if cube else [i * i for i in range(1, 20) if lo < i * i < hi]

    return Q(f'Write down all the {"cube" if cube else "square"} numbers between {lo} and {hi}.', ', '.join(map(str, xs)), ', '.join(f'{round(x ** (1 / 3)) if cube else int(x ** .5)}{"³" if cube else "²"} = {x}' for x in xs) + '.', 'Square and cube numbers')

    return Q(f'{a + x} − {v} = {a}. Find the value of {v}.', x, f'{v} = {a + x} − {a} = {x}.', 'One-step equations')


def alg_mul(r):
    v = r.choice(LET); k = r.randint(2, 9); x = r.randint(2, 12); c = r.randint(1, 20)
    rhs = r.choice([f'{k * x + c} − {c}', f'{k * x - c} + {c}']) if k * x > c else f'{k * x + c} − {c}'
    return Q(f'{k}{v} = {rhs}. Find the value of {v}.', x, f'{k}{v} = {k * x}, so {v} = {k * x} ÷ {k} = {x}.', 'Equations with a multiple')


def alg_collect(r):
    v = r.choice(LET); a = r.randint(2, 7); b = r.randint(1, 5); x = r.randint(2, 12); plus = r.random() < .5 or a <= b
    k = a + b if plus else a - b; bs = '' if b == 1 else str(b)
    return Q(f'{a}{v} {"+" if plus else "−"} {bs}{v} = {k * x}. Find the value of {v}.', x, f'{a}{v} {"+" if plus else "−"} {bs}{v} = {k if k > 1 else ""}{v}, so {v} = {k * x} ÷ {k} = {x}.', 'Collecting like terms')


def alg_div(r):
    v = r.choice(LET); a = r.randint(3, 12); b = r.randint(3, 12); t = r.randint(0, 2)
    if t == 0: return Q(f'{a * b}/{v} = {b}. Find the value of {v}.', a, f'{a * b} ÷ {v} = {b}, so {v} = {a * b} ÷ {b} = {a}.', 'Equations with division')
    if t == 1: return Q(f'{v}/{a} = {b}. Find the value of {v}.', a * b, f'{v} ÷ {a} = {b}, so {v} = {b} × {a} = {a * b}.', 'Equations with division')
    return Q(f'{a * b}/{a} = {v}. Find the value of {v}.', b, f'{a * b} ÷ {a} = {b}.', 'Equations with division')


def alg_subst(r):
    a, b = r.randint(2, 9), r.randint(2, 9); p, q = r.randint(2, 6), r.randint(2, 5); t = r.randint(0, 3)
    if t == 0 and p * a > q * b: e, v = f'{p}a − {q}b', p * a - q * b
    elif t == 1: e, v = f'{p}a + {q}b', p * a + q * b
    elif t == 2: e, v = f'ab + {p}', a * b + p
    else: e, v = f'a² + {q}b', a * a + q * b
    return Q(f'If a = {a} and b = {b}, find the value of {e}.', v, f'Replace a with {a} and b with {b}: the value is {v}.', 'Substitution')

def _seq(xs, f=str): return ', '.join(f(x) for x in xs) + ', ____, ____'


def seq_arith(r):

    return Q('Write the next two numbers: ' + _seq(xs[:4]), f'{xs[4]}, {xs[5]}', f'The rule is {"add" if d > 0 else "subtract"} {abs(d)}.', 'Next two numbers')


def seq_grow(r):
    a = r.randint(1, 40); d0 = r.randint(1, 6); dd = r.choice([1, 2, 3]); xs = [a]; d = d0
    for i in range(6): xs.append(xs[-1] + d); d += dd
    return Q('Write the next two numbers: ' + _seq(xs[:5]), f'{xs[5]}, {xs[6]}', f'The gaps grow by {dd} each time: +{d0}, +{d0 + dd}, +{d0 + 2 * dd}, +{d0 + 3 * dd}, then +{d0 + 4 * dd} and +{d0 + 5 * dd}.', 'Growing gaps')


def seq_shrink(r):
    d0 = r.randint(8, 14); a = r.randint(90, 160); xs = [a]; d = d0
    for i in range(6): xs.append(xs[-1] - d); d -= 1
    return Q('Write the next two numbers: ' + _seq(xs[:4]), f'{xs[4]}, {xs[5]}', f'The numbers go down by {d0}, then {d0 - 1}, then {d0 - 2}, so next subtract {d0 - 3} and then {d0 - 4}.', 'Shrinking gaps')

def _mach(r):
    k = r.randint(2, 9); c = r.randint(1, 12); plus = r.random() < .6
    return k, c, plus, f'× {k}, then {"+" if plus else "−"} {c}'


def fun_out(r):
    k, c, plus, d = _mach(r); x = r.randint(6, 60)
    return Q(f'A number machine works like this: IN → {d} → OUT. What comes out when {x} goes in?', k * x + (c if plus else -c), f'{x} × {k} = {k * x}. {k * x} {"+" if plus else "−"} {c} = {k * x + (c if plus else -c)}.', 'Finding the output')


def fun_in(r):
    k, c, plus, d = _mach(r); x = r.randint(4, 40); y = k * x + (c if plus else -c)
    return Q(f'A number machine works like this: IN → {d} → OUT. The number {y} comes out. What number went in?', x, f'Work backwards: {y} {"−" if plus else "+"} {c} = {k * x}, then {k * x} ÷ {k} = {x}.', 'Finding the input')


def fun_table(r):
    k = r.randint(2, 12); c = r.randint(0, 9); xs = sorted(r.sample(range(2, 15), 4)); i = r.randint(0, 3)
    rule = f'{k} × ▽' + (f' + {c}' if c else '')
    tab = [['▽'] + [str(x) for x in xs], [rule] + ['x' if j == i else str(k * x + c) for j, x in enumerate(xs)]]
    return Q('What is the missing number x in this table?', k * xs[i] + c, f'{k} × {xs[i]}' + (f' + {c}' if c else '') + f' = {k * xs[i] + c}.', 'Tables of values', table=tab)

    k = r.randint(2, 9); a, b = r.sample(range(2, 12), 2); c = r.randint(1, 9)
    return Q(f'A machine multiplies by a number and then adds {c}. It changes {a} into {a * k + c} and {b} into {b * k + c}. What number does it multiply by?', k, f'{a * k + c} − {c} = {a * k}, and {a * k} ÷ {a} = {k}. Check: {b} × {k} + {c} = {b * k + c}.', 'Finding the rule')

(The machine adds {k - 1} lots of the number and takes away {c}, so {k - 1} lots must equal {c}.)', 'Number that stays the same')

    return Q(f'A number machine works like this: IN → {"+" if plus else "−"} {c}, then × {k} → OUT. The number {y} comes out. What number went in?', x, f'Work backwards: {y} ÷ {k} = {y // k}, then {y // k} {"−" if plus else "+"} {c} = {x}.', 'Finding the input')

OPS = {'+': lambda a, b: a + b, '−': lambda a, b: a - b, '×': lambda a, b: a * b, '÷': lambda a, b: F(a, b)}


def sgn_ops2(r):
    while True:
        a, b, c = r.randint(2, 12), r.randint(2, 9), r.randint(2, 9); o1, o2 = r.choice('+−×÷'), r.choice('+−×÷')
        v = OPS[o2](OPS[o1](F(a), F(b)), F(c))
        if v.denominator != 1 or v < 0: continue
        sols = [(p, q) for p, q in iprod('+−×÷', repeat=2) if OPS[q](OPS[p](F(a), F(b)), F(c)) == v]
        if len(sols) == 1: break
    return Q(f'Put +, −, × or ÷ in each gap to make this correct:   ({a} ____ {b}) ____ {c} = {v}', f'{o1} and {o2}', f'({a} {o1} {b}) {o2} {c} = {OPS[o1](F(a), F(b))} {o2} {c} = {v}.', 'Inserting signs')


def sgn_eq(r):
    while True:
        a, b, c, d = [r.randint(2, 12) for _ in range(4)]
        sols = [(p, q) for p, q in iprod('+−×÷', repeat=2) if OPS[p](F(a), F(b)) == OPS[q](F(c), F(d))]
        if len(sols) == 1 and len({a, b, c, d}) >= 3 and sols[0][0] != sols[0][1]: break
    p, q = sols[0]; v = OPS[p](F(a), F(b))
    return Q(f'Put +, −, × or ÷ in each gap to make this correct:   {a} ____ {b} = {c} ____ {d}', f'{p} and {q}', f'{a} {p} {b} = {fr(v)} and {c} {q} {d} = {fr(v)}.', 'Inserting signs')

       ('5 miles', '5 km', 0, 'further', '5 miles is about 8 km.'), ('1 inch', '1 cm', 0, 'longer', '1 inch is about 2.5 cm.'), ('100 g', '1 ounce', 0, 'heavier', '1 ounce is about 28 g.'),
       ('1 gallon', '3 litres', 0, 'more', '1 gallon is about 4.5 litres.'), ('6 feet', '2 metres', 1, 'longer', '6 feet is about 1.8 m.'), ('30 cm', '1 foot', 1, 'longer', '1 foot is about 30.5 cm, so it is slightly longer.'),
       ('1 tonne', '500 kg', 0, 'heavier', '1 tonne = 1,000 kg.'), ('8 km', '4 miles', 0, 'further', '4 miles is about 6.4 km.')]


def mea_imp(r):
    a, b, w, adj, why = r.choice(IMP)
    if r.random() < .5: return Q(f'Which is {adj}: {a} or {b}?', [a, b][w], why, 'Imperial and metric')
    return Q(f'Which is {adj}: {b} or {a}?', [a, b][w], why, 'Imperial and metric')


def mea_approx(r):
    I, M, f, pl = r.choice([('oz', 'g', D(25), 'ounce'), ('inches', 'cm', D('2.5'), 'inch'), ('miles', 'km', D('1.6'), 'mile'), ('gallons', 'litres', D('4.5'), 'gallon'), ('lb', 'g', D(450), 'pound'), ('feet', 'cm', D(30), 'foot')])
    k = r.randint(2, 40) * (2 if f in (D('2.5'), D('4.5')) else 1)
    if r.random() < .7: return Q(f'Use the approximation 1 {pl} = {dec(f)} {M}. Convert {k} {I} to {M}.', f'{dec(k * f)} {M}', f'{k} × {dec(f)} = {dec(k * f)}.', 'Imperial to metric')
    return Q(f'Use the approximation 1 {pl} = {dec(f)} {M}. Convert {dec(k * f)} {M} to {I}.', f'{k} {I}', f'{dec(k * f)} ÷ {dec(f)} = {k}.', 'Metric to imperial')

DAYS = ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday', 'Sunday']


def tim_between(r):
    a = r.randint(20 * 60, 23 * 60 + 55); b = r.randint(5, 5 * 60); i = r.randint(0, 5); d = 1440 - a + b
    return Q(f'How many minutes are there from {t12(a)} on {DAYS[i]} to {t12(b)} on {DAYS[i + 1]}?', f'{d} minutes', f'To midnight: {1440 - a} minutes. After midnight: {b} minutes. {1440 - a} + {b} = {d}.', 'Time intervals')


def tim_walk(r):
    a = r.randint(7 * 60, 17 * 60); d = r.randint(13, 58)
    while (a % 60) + d < 60: d += 7
    return Q(f'It takes me {d} minutes to walk home. If I leave at {t12(a)}, at what time will I get home?', t12(a + d), f'{60 - a % 60} minutes takes me to {t12(a + 60 - a % 60)}, then {d - (60 - a % 60)} more minutes: {t12(a + d)}.', 'Finishing times')


def tim_wait(r):
    a = r.randint(7 * 60, 18 * 60); late = r.randint(3, 14); gap = r.randint(25, 55); A = r.choice(['Mrs Late', 'Mr Rush', 'Miss Hurry', 'Dr Dash'])
    return Q(f'{A} arrives {late} minutes late for the {t24(a)} train. How long will {A} have to wait for the next train at {t24(a + gap)}?', f'{gap - late} minutes',
             f'{A} arrives at {t24(a + late)}. From {t24(a + late)} to {t24(a + gap)} is {gap - late} minutes.', 'Waiting times')


def tim_watch(r):
    per, nm = r.choice([(15, 'a quarter of a minute'), (30, 'half a minute'), (20, 'one third of a minute'), (120, '2 minutes'), (90, '1½ minutes')]); st = r.choice([8, 9, 10, 12]) * 60; h = r.choice([4, 6, 8, 10, 12]); w = r.choice(['loses', 'gains'])
    tot = per * h
    if tot % 60: h = 8 if (per * 8) % 60 == 0 else 12; tot = per * h
    sh = st + h * 60 + (tot // 60 if w == 'gains' else -tot // 60)
    return Q(f'My watch {w} {nm} every hour. I put it right at {t12(st)}. What time will my watch show when the real time is {t12(st + h * 60)}?', t12(sh),
             f'{h} hours pass, so it {w} {h} × {nm} = {tot // 60} minute{"s" if tot != 60 else ""}. It shows {t12(sh)}.', 'Fast and slow clocks')


def tim_show(r):
    st = 60 * r.randint(13, 19) + r.choice([0, 15, 30, 45]); a = r.randint(50, 105); b = r.randint(8, 25)
    return Q(f'A show starts at {t12(st)}. The first half lasts {dur(a).replace(" h", " hour").replace(" min", " minutes")}, then there is an interval of {b} minutes. At what time does the second half begin?', t12(st + a + b),
             f'{dur(a)} + {b} min = {dur(a + b)}. {t12(st)} + {dur(a + b)} = {t12(st + a + b)}.', 'Finishing times')


def tim_lessons(r):
    k = r.choice([4, 5, 6]); L = r.choice([30, 35, 40, 45, 50]); B = r.choice([15, 20, 25]); j = r.randint(1, k - 1); end = 60 * r.randint(12, 15) + r.choice([0, 10, 20, 25, 30, 50])
    start = end - k * L - B; t = r.randint(0, 2); o = ['1st', '2nd', '3rd', '4th', '5th', '6th']
    stem = f'A school has {k} lessons, each {L} minutes long, with a break of {B} minutes after the {o[j - 1]} lesson. The {o[k - 1]} lesson ends at {t24(end)}. '
    if t == 0: return Q(stem + 'At what time does the 1st lesson begin?', t24(start), f'Total time: {k} × {L} + {B} = {k * L + B} minutes = {dur(k * L + B)}. {t24(end)} − {dur(k * L + B)} = {t24(start)}.', 'Timetables')
    if t == 1: return Q(stem + 'At what time does the break begin?', t24(start + j * L), f'The 1st lesson begins at {t24(start)}. The break begins after {j} lesson{"s" if j > 1 else ""}: {t24(start)} + {dur(j * L)} = {t24(start + j * L)}.', 'Timetables')
    return Q(stem + f'At what time does the {o[k - 1]} lesson begin?', t24(end - L), f'{t24(end)} − {L} minutes = {t24(end - L)}.', 'Timetables')


STN = ['Ashby', 'Brookfield', 'Carlow', 'Denton', 'Elmwood', 'Fairholt', 'Greyford', 'Hartwell', 'Inglemere', 'Kelby', 'Linton', 'Marsham']


def _trains(r, nt=2):
    st = r.sample(STN, 5); cols = []
    for t in range(nt):
        m = 60 * r.randint(6, 18) + r.randint(0, 59); ts = [m]
        for _ in range(4): ts.append(ts[-1] + r.randint(6, 24))
        cols.append(ts)
    tab = [['Station'] + [f'Train {"ABCD"[t]}' for t in range(nt)]] + [[st[i]] + [t24(c[i]) for c in cols] for i in range(5)]
    return st, cols, tab


def tim_train(r):
    st, cols, tab = _trains(r); t = r.randint(0, 2)
    if t == 0:
        c = r.randint(0, 1); i, j = sorted(r.sample(range(5), 2)); d = cols[c][j] - cols[c][i]
        return Q(f'Look at the train timetable. How long does Train {"AB"[c]} take to travel from {st[i]} to {st[j]}?', f'{dur(d)}', f'{t24(cols[c][i])} to {t24(cols[c][j])} is {dur(d)}.', 'Train timetables', table=tab)
    if t == 1:
        da, db = cols[0][4] - cols[0][0], cols[1][4] - cols[1][0]
        if da == db: return tim_train(r)
        return Q(f'Look at the train timetable. Which train is faster for the whole journey from {st[0]} to {st[4]}, and by how many minutes?', f'Train {"A" if da < db else "B"}, by {abs(da - db)} minutes',
                 f'Train A takes {dur(da)} and Train B takes {dur(db)}.', 'Train timetables', table=tab)
    c = r.randint(0, 1); i = r.randint(0, 2); walk = r.randint(8, 25)
    return Q(f'Look at the train timetable. I walk for {walk} minutes to reach {st[i]} station and want to catch Train {"AB"[c]}. What is the latest time I can leave home?', t24(cols[c][i] - walk),
             f'Train {"AB"[c]} leaves {st[i]} at {t24(cols[c][i])}. {t24(cols[c][i])} − {walk} minutes = {t24(cols[c][i] - walk)}.', 'Train timetables', table=tab)


ZONES = [('Paris', 1), ('Athens', 2), ('Nairobi', 3), ('Dubai', 4), ('Karachi', 5), ('Dhaka', 6), ('Bangkok', 7), ('Beijing', 8), ('Tokyo', 9), ('Sydney', 10), ('New York', -5), ('Chicago', -6), ('Denver', -7), ('Los Angeles', -8), ('Rio de Janeiro', -3)]


def tim_zones(r):
    zs = r.sample(ZONES, 4); tab = [['City', 'Time compared with London']] + [[c, f'{abs(h)} hour{"s" if abs(h) > 1 else ""} {"ahead" if h > 0 else "behind"}'] for c, h in zs]
    m = 60 * r.randint(6, 20) + r.choice([0, 15, 30, 45]); t = r.randint(0, 2); (c1, h1), (c2, h2) = zs[0], zs[1]
    dd = lambda x: t12(x) + (' the day before' if x < 0 else ' the next day' if x >= 1440 else '')
    if t == 0: return Q(f'Use the time chart. When it is {t12(m)} in London, what time is it in {c1}?', dd(m + 60 * h1), f'{c1} is {abs(h1)} hour{"s" if abs(h1) > 1 else ""} {"ahead" if h1 > 0 else "behind"}: {t12(m + 60 * h1)}.', 'Time zones', table=tab)
    if t == 1: return Q(f'Use the time chart. When it is {t12(m)} in {c1}, what time is it in London?', dd(m - 60 * h1), f'London is {abs(h1)} hour{"s" if abs(h1) > 1 else ""} {"behind" if h1 > 0 else "ahead of"} {c1}: {t12(m - 60 * h1)}.', 'Time zones', table=tab)
    return Q(f'Use the time chart. When it is {t12(m)} in {c1}, what time is it in {c2}?', dd(m - 60 * h1 + 60 * h2), f'London time is {t12(m - 60 * h1)}. {c2} is {abs(h2)} hour{"s" if abs(h2) > 1 else ""} {"ahead" if h2 > 0 else "behind"}: {t12(m - 60 * h1 + 60 * h2)}.', 'Time zones', table=tab)


def tim_cook(r):
    per = r.choice([40, 45, 50, 55, 60, 35]); ex = r.choice([15, 20, 25, 30]); w = D(r.choice(['1.5', '2', '2.5', '3', '3.5', '4', '4.5'])); fd = r.choice(['Roast chicken', 'A joint of beef', 'A nut roast', 'A turkey crown'])
    tot = int(per * w) + ex
    if (per * w) % 1: return tim_cook(r)
    return Q(f'{fd} must be cooked for {per} minutes per kg and then for an extra {ex} minutes. What is the cooking time for one that weighs {dec(w)} kg? Give your answer in hours and minutes.', dur(tot),
             f'{dec(w)} × {per} = {int(per * w)} minutes. {int(per * w)} + {ex} = {tot} minutes = {dur(tot)}.', 'Cooking times')


def tim_clock(r):
    h = r.randint(1, 11); ang = 30 * h if h <= 6 else 360 - 30 * h
    return Q(f'What is the size of the smaller angle between the hands of a clock at {h} o\'clock?', f'{ang}°', f'Each hour is 360° ÷ 12 = 30°. The hands are {h if h <= 6 else 12 - h} hour space{"s" if (h if h <= 6 else 12 - h) > 1 else ""} apart: {ang}°.', 'Clock angles')
