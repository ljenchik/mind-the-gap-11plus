def names(r, k): return r.sample(NAMES, k)


def n(x): return f'{x:,}'


def dec(x):
    x = D(str(x))
    s = format(x.normalize(), 'f')
    if '.' in s:
        a, b = s.split('.')
        return f'{int(a):,}.{b}'
    return f'{int(s):,}'


def money(p):
    p = int(p)
    return f'£{p // 100:,}.{p % 100:02d}'


def pm(p): return f'{p}p' if p < 100 else money(p)


def fr(x):
    x = F(x)
    if x.denominator == 1: return str(x.numerator)
    w = x.numerator // x.denominator
    rem = x - w
    return f'{w} {rem.numerator}/{rem.denominator}' if w else f'{x.numerator}/{x.denominator}'


def raw(a, b): return f'{a}/{b}'


def t12(m):
    m %= 1440
    h, mi = divmod(m, 60)
    suf = 'a.m.' if h < 12 else 'p.m.'
    h12 = h % 12 or 12
    return f'{h12}:{mi:02d} {suf}'


def t24(m):
    m %= 1440
    return f'{m // 60:02d}:{m % 60:02d}'


def dur(m):
    h, mi = divmod(m, 60)
    if h and mi: return f'{h} h {mi} min'
    return f'{h} h' if h else f'{mi} min'


import re
def fx(t): return re.sub(r'(?<![\w.])-(?=\d)', '−', t.replace('a.m..', 'a.m.').replace('p.m..', 'p.m.'))


def Q(text, ans, sol, typ, fig=None, table=None, alt=None):
    return dict(text=fx(text), ans=fx(str(ans)), sol=fx(sol), type=typ, fig=fig, table=table, alt=fx(alt) if alt else alt)


def P(intro, parts, skill, fig=None, table=None):
    """paper question: parts = [(text, ans, sol)]"""
    return dict(intro=[fx(x) for x in (intro if isinstance(intro, list) else [intro])], parts=[(fx(a), fx(str(b)), fx(c)) for a, b, c in parts],
                skill=skill, fig=fig, table=table)


def primes_upto(k): return [p for p in range(2, k + 1) if all(p % q for q in range(2, int(p ** .5) + 1))]


def factors(k): return [d for d in range(1, k + 1) if k % d == 0]


def pfactors(k):
    out, p = [], 2
    while k > 1:
        while k % p == 0: out.append(p); k //= p
        p += 1
    return out

def _save(fig, path):
    fig.savefig(path, dpi=150, bbox_inches='tight', pad_inches=0.08)
    plt.close(fig)


def fig_tri(path, pts, W, H):
    fig, ax = plt.subplots(figsize=(W * .42, H * .42))
    for x in range(W + 1): ax.plot([x, x], [0, H], color='#9aa', lw=.7)
    for y in range(H + 1): ax.plot([0, W], [y, y], color='#9aa', lw=.7)
    ax.add_patch(Polygon(pts, closed=True, fc='#cfd8e6', ec='black', lw=1.8))
    ax.set_xlim(-.1, W + .1); ax.set_ylim(-.1, H + .1); ax.set_aspect('equal'); ax.axis('off')
    _save(fig, path)


def fig_lshape(path, a, b, c, d, unit):
    fig, ax = plt.subplots(figsize=(3.4, 3.0))
    pts = [(0, 0), (a, 0), (a, d), (c, d), (c, b), (0, b)]

    ax.text(a + .04 * m, d / 2, f'{d} {unit}', ha='left', va='center', fontsize=11)

def fig_numline(path, lo, step, nt, every, arrows):
    """lo: start (float), step: gap between small ticks, nt: number of gaps, every: label every k ticks"""
    fig, ax = plt.subplots(figsize=(6.2, 1.3))
    hi = lo + step * nt
    ax.plot([lo, hi], [0, 0], color='black', lw=1.5)
    for i in range(nt + 1):
        x = lo + i * step
        big = i % every == 0
        ax.plot([x, x], [0, .35 if big else .2], color='black', lw=1.2)
        if big: ax.text(x, -.25, dec(round(x, 6)), ha='center', va='top', fontsize=10)
    for lab, x in arrows.items():
        ax.annotate(lab, xy=(x, .4), xytext=(x, 1.25), ha='center', fontsize=11, fontweight='bold',
                    arrowprops=dict(arrowstyle='-|>', color='black', lw=1.3))
    ax.set_xlim(lo - step, hi + step); ax.set_ylim(-.9, 1.7); ax.axis('off')
    _save(fig, path)


def fig_pie(path, labels, angles, show=True):
    fig, ax = plt.subplots(figsize=(3.6, 3.6))
    greys = ['#ffffff', '#d5dbe5', '#aab4c4', '#eef1f6', '#c2cad8', '#e0e4ec']
    st = 90
    for i, (lab, a) in enumerate(zip(labels, angles)):
        ax.add_patch(Wedge((0, 0), 1, st - a, st, fc=greys[i % 6], ec='black', lw=1.4))
        mid = math.radians(st - a / 2)
        txt = f'{lab}\n{a}°' if show else lab
        ax.text(.66 * math.cos(mid), .66 * math.sin(mid), txt, ha='center', va='center', fontsize=8.5)
        st -= a
    ax.set_xlim(-1.05, 1.05); ax.set_ylim(-1.05, 1.05); ax.set_aspect('equal'); ax.axis('off')
    _save(fig, path)

    ax.grid(axis='y', color='#888', lw=.5); ax.set_axisbelow(True)
    ax.set_ylabel(ylabel, fontsize=10); ax.tick_params(labelsize=9)
    _save(fig, path)

def fig_travel(path, lines, t0, hours, dmax, dstep):
    """lines: [(label, [(minutes_from_t0, km), ...])]; x axis in hours from t0 (minutes since midnight)"""
    fig, ax = plt.subplots(figsize=(5.6, 3.5))
    sty = ['-', '--']
    for i, (lab, pts) in enumerate(lines):
        ax.plot([p[0] for p in pts], [p[1] for p in pts], sty[i % 2], color='black', lw=1.8, label=lab)
    ax.set_xticks(range(0, hours * 60 + 1, 30))
    ax.set_xticklabels([t24(t0 + m) if m % 60 == 0 else '' for m in range(0, hours * 60 + 1, 30)], fontsize=8)
    ax.set_yticks(range(0, dmax + 1, dstep)); ax.tick_params(labelsize=8)
    ax.set_xlim(0, hours * 60); ax.set_ylim(0, dmax); ax.grid(color='#999', lw=.5)
    ax.set_xlabel('Time', fontsize=9); ax.set_ylabel('Distance (km)', fontsize=9)
    if len(lines) > 1 or lines[0][0]: ax.legend(fontsize=8, loc='upper left')
    _save(fig, path)

def render_fig(spec):
    os.makedirs(FIGDIR, exist_ok=True)
    h = hashlib.md5(repr(spec).encode()).hexdigest()[:10]
    path = f'{FIGDIR}/{spec[0]}_{h}.png'
    if not os.path.exists(path): FIGS[spec[0]](path, *spec[1:])
    return path

ONES = 'zero one two three four five six seven eight nine ten eleven twelve thirteen fourteen fifteen sixteen seventeen eighteen nineteen'.split()
TENS = '_ _ twenty thirty forty fifty sixty seventy eighty ninety'.split()
def words(k):
    if k < 20: return ONES[k]
    if k < 100: return TENS[k // 10] + ('-' + ONES[k % 10] if k % 10 else '')
    if k < 1000: return ONES[k // 100] + ' hundred' + (' and ' + words(k % 100) if k % 100 else '')
    return words(k // 1000) + ' thousand' + ((' and ' if k % 1000 < 100 else ' ') + words(k % 1000) if k % 1000 else '')
