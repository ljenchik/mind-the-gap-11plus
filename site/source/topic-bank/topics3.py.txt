def mon_rides(r):
    while True:
        ps = [5 * r.randint(22, 70) for _ in range(4)]
        sums = [a + b for a, b in combinations(ps, 2)]
        if len(set(sums)) == 6 and len(set(ps)) == 4: break
    nm = r.sample(['Galaxy', 'Big Wheel', 'Laser', 'Rocket', 'Twister', 'Waltzer', 'Log Flume', 'Dodgems'], 4); tab = [['Ride', 'Cost']] + [[a, money(b)] for a, b in zip(nm, ps)]
    t = r.randint(0, 2)
    if t == 0:
        i, j = r.sample(range(4), 2); A = names(r, 1)[0]
        return Q(f'Look at the ride prices. {A} went on two different rides and had {money(1000 - ps[i] - ps[j])} change from £10. Which two rides did {A} go on?', ' and '.join(sorted([nm[i], nm[j]])),
                 f'{A} spent £10.00 − {money(1000 - ps[i] - ps[j])} = {money(ps[i] + ps[j])}. {money(ps[i])} + {money(ps[j])} = {money(ps[i] + ps[j])}.', 'Price lists', table=tab)
    if t == 1: return Q('Look at the ride prices. How much would it cost to go on all four rides once?', money(sum(ps)), ' + '.join(money(p) for p in ps) + f' = {money(sum(ps))}.', 'Price lists', table=tab)
    if sum(ps) % 4: return mon_rides(r)
    return Q('Look at the ride prices. What is the mean (average) price of a ride?', money(sum(ps) // 4), f'Total {money(sum(ps))} ÷ 4 = {money(sum(ps) // 4)}.', 'Price lists', table=tab)

def par_width(r):
    L, W = r.randint(7, 40), r.randint(2, 20); u = r.choice(['cm', 'm'])
    if L <= W: L += W
    return Q(f'A rectangle has a perimeter of {2 * (L + W)} {u} and a length of {L} {u}. What is its width?', f'{W} {u}', f'Half the perimeter is {L + W} {u}. {L + W} − {L} = {W} {u}.', 'Perimeter')


def par_length(r):
    W = D(r.choice(['2', '3', '4', '5', '1.5', '2.5', '3.5', '6', '8', '0.5'])); L = D(r.choice(['3', '4', '7', '9', '1.2', '2.4', '12', '6.5', '15', '4.5', '1.3'])); u = r.choice(['cm', 'm'])
    if r.random() < .5: return Q(f'A rectangle has an area of {dec(L * W)} {u}² and a width of {dec(W)} {u}. What is its length?', f'{dec(L)} {u}', f'{dec(L * W)} ÷ {dec(W)} = {dec(L)}.', 'Area')
    return Q(f'What is the area of a rectangle {dec(L)} {u} long and {dec(W)} {u} wide?', f'{dec(L * W)} {u}²', f'{dec(L)} × {dec(W)} = {dec(L * W)}.', 'Area')


def par_square(r):
    s = r.randint(3, 25); u = r.choice(['cm', 'm']); t = r.randint(0, 2)
    if t == 0: return Q(f'A square has a perimeter of {4 * s} {u}. What is its area?', f'{s * s} {u}²', f'Each side is {4 * s} ÷ 4 = {s} {u}. {s} × {s} = {s * s} {u}².', 'Squares')
    if t == 1: return Q(f'A square has an area of {s * s} {u}². What is its perimeter?', f'{4 * s} {u}', f'{s} × {s} = {s * s}, so each side is {s} {u}. 4 × {s} = {4 * s} {u}.', 'Squares')
    L = r.randint(s + 1, 2 * s - 1); W = 2 * s - L
    return Q(f'A square has the same perimeter as a rectangle {L} {u} long and {W} {u} wide. What is the length of one side of the square?', f'{s} {u}', f'Rectangle perimeter: 2 × ({L} + {W}) = {4 * s} {u}. {4 * s} ÷ 4 = {s} {u}.', 'Squares')


def par_ktimes(r):
    k = r.choice([2, 3, 4, 5]); W = r.choice([r.randint(2, 15), D(r.randint(3, 19)) / 2]); L = k * W; P = 2 * (L + W); u = r.choice(['cm', 'm']); t = r.randint(0, 2)
    stem = f'The perimeter of a rectangle is {dec(P)} {u}. Its length is {k} times its width. '
    why = f'Length + width is half the perimeter, {dec(P / 2)} {u}, and that is {k + 1} equal parts. One part (the width) is {dec(W)} {u}; the length is {dec(L)} {u}.'
    if t == 0: return Q(stem + 'What is the length?', f'{dec(L)} {u}', why, 'Length is a multiple of the width')
    if t == 1: return Q(stem + 'What is the width?', f'{dec(W)} {u}', why, 'Length is a multiple of the width')
    return Q(stem + 'What is the area?', f'{dec(L * W)} {u}²', why + f' Area = {dec(L)} × {dec(W)} = {dec(L * W)} {u}².', 'Length is a multiple of the width')


def par_path(r):
    L, W = 5 * r.randint(3, 12), 5 * r.randint(1, 6); w = r.choice([1, 2, 3, 5, 10]); t = r.randint(0, 2)
    if L <= W: L += W
    th = r.choice(['swimming pool', 'lawn', 'flower bed', 'playground'])
    stem = f'A rectangular {th} is {L} m long and {W} m wide. It has a path {w} m wide all the way round it. '
    big = (L + 2 * w) * (W + 2 * w)
    if t == 0: return Q(stem + f'What is the area of the path?', f'{n(big - L * W)} m²', f'Whole area: {L + 2 * w} × {W + 2 * w} = {n(big)} m². {th.capitalize()}: {L} × {W} = {n(L * W)} m². Path: {n(big)} − {n(L * W)} = {n(big - L * W)} m².', 'Paths and borders')
    if t == 1: return Q(stem + f'What is the area of the {th} and path together?', f'{n(big)} m²', f'The whole rectangle is {L} + {2 * w} = {L + 2 * w} m by {W} + {2 * w} = {W + 2 * w} m. {L + 2 * w} × {W + 2 * w} = {n(big)} m².', 'Paths and borders')
    return Q(stem + 'What is the perimeter of the outside edge of the path?', f'{2 * (L + W + 4 * w)} m', f'The outside rectangle is {L + 2 * w} m by {W + 2 * w} m. 2 × ({L + 2 * w} + {W + 2 * w}) = {2 * (L + W + 4 * w)} m.', 'Paths and borders')

def tri_grid(r):
    b = r.randint(2, 8); h = r.randint(2, 6)
    while (b * h) % 2 and r.random() < .5: b += 1
    x0 = r.randint(0, 2); ax = r.randint(0, x0 + b + 1); Wd = max(x0 + b, ax) + 1
    pts = [(x0, 1), (x0 + b, 1), (ax, 1 + h)]
    return Q('Find the area of this triangle. Each square on the grid is 1 cm².', f'{dec(D(b * h) / 2)} cm²', f'Base {b} cm, height {h} cm. ½ × {b} × {h} = {dec(D(b * h) / 2)} cm².', 'Triangles on a grid',
             fig=('tri', pts, Wd, h + 2), alt=f'[Triangle drawn on a 1 cm square grid with corners at {pts[0]}, {pts[1]} and {pts[2]}.]')

def coo_robot(r):
    x, y = r.randint(0, 5), r.randint(0, 5); f1, f2 = r.randint(1, 6), r.randint(1, 6); turn = r.choice(['RIGHT', 'LEFT']); d0 = r.choice(['north', 'east'])

# ---------- 27 Averages ----------
def _set(r, k, lo=1, hi=30): return [r.randint(lo, hi) for _ in range(k)]


def avg_mean(r):
    k = r.choice([3, 4, 5, 6])
    while True:
        xs = _set(r, k)
        if (2 * sum(xs)) % k == 0: break
    return Q(f'Find the mean of these numbers: {", ".join(map(str, xs))}', dec(D(sum(xs)) / k), f'Total: {sum(xs)}. {sum(xs)} ÷ {k} = {dec(D(sum(xs)) / k)}.', 'Mean')


def avg_median(r):
    k = r.choice([4, 5, 6, 7]); xs = _set(r, k, 1, 60); s = sorted(xs)
    med = D(s[k // 2]) if k % 2 else D(s[k // 2 - 1] + s[k // 2]) / 2
    return Q(f'Find the median of these numbers: {", ".join(map(str, xs))}', dec(med), f'In order: {", ".join(map(str, s))}. ' + (f'The middle number is {dec(med)}.' if k % 2 else f'The two middle numbers are {s[k // 2 - 1]} and {s[k // 2]}; halfway between them is {dec(med)}.'), 'Median')


def avg_mode(r):
    while True:
        xs = _set(r, r.choice([6, 7, 8]), 3, 19); c = sorted(((xs.count(x), x) for x in set(xs)), reverse=True)
        if c[0][0] >= 2 and (len(c) == 1 or c[1][0] < c[0][0]): break
    return Q(f'What is the mode of these numbers? {", ".join(map(str, xs))}', c[0][1], f'{c[0][1]} appears {c[0][0]} times, more often than any other number.', 'Mode')


def avg_range(r):
    xs = _set(r, r.choice([5, 6, 7]), 3, 99)
    return Q(f'What is the range of these numbers? {", ".join(map(str, xs))}', max(xs) - min(xs), f'Largest − smallest = {max(xs)} − {min(xs)} = {max(xs) - min(xs)}.', 'Range')


def avg_changed(r):
    ns = names(r, 5)
    while True:
        xs = _set(r, 5, 10, 40); i = r.randint(0, 4); new = xs[i] + r.choice([-10, -5, 5, 10, -1, 1])
        ys = xs[:]; ys[i] = new
        if sum(xs) % 5 == 0 and sum(ys) % 5 == 0 and new > 0: break
    tab = [['Name'] + ns, ['Score'] + [str(x) for x in xs]]; t = r.randint(0, 2)
    stem = f'Here are the scores in a test. {ns[i]}\'s test was wrongly marked: the score should have been {new}, not {xs[i]}. '
    if t == 0: return Q(stem + 'What is the correct mean score?', sum(ys) // 5, f'New total: {sum(xs)} {"+" if new > xs[i] else "−"} {abs(new - xs[i])} = {sum(ys)}. {sum(ys)} ÷ 5 = {sum(ys) // 5}.', 'Correcting a mistake', table=tab)
    if t == 1: return Q(stem + 'What is the correct median score?', sorted(ys)[2], f'Correct scores in order: {", ".join(map(str, sorted(ys)))}. The middle one is {sorted(ys)[2]}.', 'Correcting a mistake', table=tab)
    return Q(stem + 'What is the correct range?', max(ys) - min(ys), f'{max(ys)} − {min(ys)} = {max(ys) - min(ys)}.', 'Correcting a mistake', table=tab)


def avg_next(r):
    k = r.randint(4, 9); m = r.randint(8, 40); up = r.randint(1, 4); th = r.choice([('A cricketer', 'innings', 'runs'), ('A basketball player', 'matches', 'points'), ('A quiz team', 'rounds', 'points')])
    return Q(f"{th[0]}'s mean score for {k} {th[1]} is {m} {th[2]}. What score is needed in the next one to make the mean {m + up}?", (m + up) * (k + 1) - m * k,

Total of the {k - 1}: {k - 1} × {m2} = {m2 * (k - 1)}. {dec(m1 * k)} − {m2 * (k - 1)} = {dec(last)}.', 'Mean problems')

def _pie(r):

def _p(a, b): return '0' if a == 0 else '1' if a == b else fr(F(a, b))


def prb_die(r):
    ev = [('a 6', [6]), ('a 2 or a 3', [2, 3]), ('an even number', [2, 4, 6]), ('a number greater than 4', [5, 6]), ('a prime number', [2, 3, 5]), ('a 7', []), ('a number less than 7', [1, 2, 3, 4, 5, 6]),
          ('a multiple of 3', [3, 6]), ('a number less than 3', [1, 2]), ('a square number', [1, 4]), ('a factor of 6', [1, 2, 3, 6]), ('an odd number greater than 1', [3, 5]), ('a number greater than 1', [2, 3, 4, 5, 6])]
    e, s = r.choice(ev)
    return Q(f'A fair dice numbered 1 to 6 is rolled. What is the probability of rolling {e}? Give your answer as a fraction in its simplest form (or 0 or 1).', _p(len(s), 6),
             (f'{len(s)} of the 6 equally likely numbers work ({", ".join(map(str, s))}): {len(s)}/6' + (f' = {_p(len(s), 6)}.' if _p(len(s), 6) != f'{len(s)}/6' else '.')) if s else 'It is impossible, so the probability is 0.', 'Dice')


def prb_bag(r):
    cs = r.sample(['grey', 'white', 'red', 'blue', 'green', 'yellow'], 3); v = [r.randint(1, 9) for _ in range(3)]; N = sum(v); t = r.randint(0, 3); th = r.choice(['balls', 'counters', 'beads', 'marbles'])
    stem = f'A bag contains {v[0]} {cs[0]} {th}, {v[1]} {cs[1]} {th} and {v[2]} {cs[2]} {th}. One is picked without looking. '
    if t == 0: return Q(stem + f'What is the probability that it is {cs[0]}? Give your answer as a fraction in its simplest form.', _p(v[0], N), f'{v[0]} out of {N}' + (f' = {_p(v[0], N)}.' if gcd(v[0], N) > 1 else '.'), 'Bags')
    if t == 1: return Q(stem + f'What is the probability that it is not {cs[0]}? Give your answer as a fraction in its simplest form.', _p(N - v[0], N), f'{N} − {v[0]} = {N - v[0]} are not {cs[0]}: {N - v[0]} out of {N}' + (f' = {_p(N - v[0], N)}.' if gcd(N - v[0], N) > 1 else '.'), 'Bags')
    if t == 2: return Q(stem + f'What is the probability that it is {cs[1]} or {cs[2]}? Give your answer as a fraction in its simplest form.', _p(v[1] + v[2], N), f'{v[1]} + {v[2]} = {v[1] + v[2]} out of {N}' + (f' = {_p(v[1] + v[2], N)}.' if gcd(v[1] + v[2], N) > 1 else '.'), 'Bags')
    return Q(stem + 'What is the probability that it is black?', '0', 'There are no black ones in the bag, so it is impossible: 0.', 'Bags')


def prb_cards(r):

                                                        ('a square number', lambda x: int(x ** .5) ** 2 == x), ('a number greater than 8', lambda x: x > 8), ('a factor of 12', lambda x: 12 % x == 0), ('a multiple of 4', lambda x: x % 4 == 0),
