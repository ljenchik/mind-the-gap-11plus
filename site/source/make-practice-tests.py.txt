def mixed(fr):
    fr=F(fr); w=fr.numerator//fr.denominator; r=fr-w
    return (f"{w} {r.numerator}/{r.denominator}" if w else f"{r.numerator}/{r.denominator}") if r else str(w)
def money(p):  # p in pence
    return f"£{p//100}.{p%100:02d}" if p%100 else f"£{p//100}"
def hm(m): h,mi=divmod(m%1440,60); return h,mi
def t12(m):
    h,mi=hm(m); s="am" if h<12 else "pm"; h=h%12 or 12; return f"{h}:{mi:02d} {s}"
def t24(m): h,mi=hm(m); return f"{h:02d}:{mi:02d}"
def dur(m):
    h,mi=divmod(m,60); a=[]
    if h: a.append(f"{h} hour{'s' if h>1 else ''}")
    if mi: a.append(f"{mi} minute{'s' if mi>1 else ''}")
    return " ".join(a)
