"""Hints that did not fit their question (27 Nov check). Safe to run more than once."""
import json, re
SING = {'machines': 'machine', 'hens': 'hen', 'workers': 'worker', 'printers': 'printer', 'robots': 'robot', 'cows': 'cow', 'bees': 'bee', 'taps': 'tap', 'pumps': 'pump'}
def fix(x, old):
    h = x.get('hint'); q = x['q']; n = 0
    if not h: return 0
    # work-rate hint put on recipes and other non-work questions: put back the original hint
    if h.startswith('Start with 1 ') and not re.search(r'\b(hours?|days?|minutes?|weeks?)\b', q):
        o = old.get(x['id'], {})
        x['hint'] = o.get('hint') or 'Find the amount for one person first.'
        if 'm' in o: x['m'] = o['m']
        if 'lv' in o: x['lv'] = o['lv']
        return 1
    if h == 'Find how much one worker does in one day.':
        m = re.search(r'\b(\d+) (machines|hens|workers|printers|robots|cows|bees|taps|pumps)\b', q); t = re.search(r'\b(hours?|days?|minutes?)\b', q)
        noun = SING.get(m[2], 'worker') if m else 'worker'; unit = (t[1].rstrip('s') if t else 'day')
        x['hint'] = f"Start with 1 {noun}: find how much 1 {noun} makes in 1 {unit}. Then scale up."; return 1
    if h.startswith('Add the two groups, take off the overlap') and not re.search(r'\bboth\b', q):
        x['hint'] = 'Add the parts you know, then take that from the whole (1 whole = 100%).'; return 1
    if h.startswith('Look at the tens and ones digits first'):
        m = re.search(r'(\d+) − (\d+)', q)
        if m:
            a, b = int(m[1]), int(m[2]); r = -(-b // 1000) * 1000
            if b < r <= a: x['hint'] = f"Count up from {b} to {r}, then on to {a}. Add the two jumps."; return 1
    return 0
if __name__ == '__main__':
    import sys
    old = {x['id']: x for x in json.load(open(sys.argv[1]))} if len(sys.argv) > 1 else {}
    if len(sys.argv) > 2: old.update({x['id']: x for x in json.load(open(sys.argv[2]))})
    for f in ('questions.json', 'cem.json'):
        d = json.load(open(f)); n = sum(fix(x, old) for x in d); json.dump(d, open(f, 'w'), ensure_ascii=False); print(f, n)
