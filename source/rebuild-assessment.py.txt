"""Initial assessment: fixes to A (sets 1 and 2) from the review sheet, and a new Assessment B (sets 1 and 2)
with different kinds of task in the same topic order. Every answer is computed here."""
import json, sys
from fractions import Fraction as F

P = sys.argv[1]
d = json.load(open(P))
papers = {p["id"]: p for p in d["papers"]}

def N(t, v, a=None, **kw):
    s = {"k": "n", "v": str(v)}
    part = {"t": t, "a": a if a is not None else str(v), "s": s}
    for k in ("suf", "ph"):
        if k in kw: part[k] = kw.pop(k)
    s.update(kw); return part
def FR(t, v, a, strict=False):
    s = {"k": "f", "v": f"{v.numerator}/{v.denominator}"}
    if strict: s["strict"] = True
    return {"t": t, "a": a, "s": s, "ph": "e.g. 3/4 or 1 1/2"}
def CH(t, v, opts): assert v in opts; return {"t": t, "a": v, "s": {"k": "ch", "v": v, "o": opts}}
def C(t, x, y): return {"t": t, "a": f"({x}, {y})", "s": {"k": "c", "v": [x, y]}, "ph": "(x, y)"}
def T(t, mins): return {"t": t, "a": f"{mins // 60:02d}:{mins % 60:02d}", "s": {"k": "t", "v": mins}, "ph": "e.g. 14:35"}
def mixed(f):
    w, r = divmod(f.numerator, f.denominator)
    return f"{w} {r}/{f.denominator}" if w and r else (str(w) if not r else f"{f.numerator}/{f.denominator}")
def Q(pid, n, sk, intro, parts, pic=None, table=None):
    q = {"id": pid, "n": n, "sk": sk, "intro": intro, "parts": parts, "fig": None, "table": table}
    if pic: q["pic"] = pic
    return q
ASN = ["Always true", "Sometimes true", "Never true"]

# ---------------- fixes to Assessment A ----------------
def fixA(pid, form):
    p = papers[pid]; qs = {q["n"]: q for q in p["qs"]}
    # 2: equivalent fractions accepted when the question does not ask for simplest form
    for part in qs[2]["parts"]:
        if "simplest form" in part["t"]:
            part["t"] = part["t"].replace(" Give your answer in its simplest form.", ""); part["s"].pop("strict", None)
    # 9: a simpler counting question (set 1 had the keypad; set 2 keeps the café puzzle)
    if qs[9]["intro"] and "keypad" in qs[9]["intro"][0]:
        hi, L = (5, "A, B or C") if form == 1 else (6, "X or Y")
        nl = 3 if form == 1 else 2
        qs[9]["intro"] = [f"A bike lock has a code of two digits. Each digit is from 1 to {hi}."]
        qs[9]["parts"] = [N("How many different codes are there if a digit can be used twice (like 33)?", hi * hi),
                          N("How many different codes are there if the two digits must be different?", hi * (hi - 1)),
                          N(f"A new lock uses one letter ({L}) followed by one digit from 1 to {hi}. How many codes are there?", nl * hi)]
    # 10: clearer sequences, one answer each
    if form == 1:
        qs[10]["parts"][1] = N("In this sequence the gap between one number and the next gets 2 bigger each time: 3, 5, 9, □, 23. Fill in the missing number.", 15)
        qs[10]["parts"][2] = N("In this sequence the gaps get 3 bigger each time: □, 14, 25, 39, 56. What is the first number?", 6)
        # 11b: a smaller number
        qs[11]["parts"][1] = N("Find the number put in if 25 is the output.", 14)
        assert (14 * 3 + 8) / 2 == 25
        # 12a: no algebra
        qs[12]["parts"][0] = N("Two angles of a triangle are 48° and 75°. Find the third angle.", 57, a="57°", suf="°")
        # 13b: a translation instead of a rotation; A is (-3, -1)
        qs[13]["parts"][1] = C("Triangle ABC is moved 4 squares to the right and 3 squares up. Give the coordinates of the new position of point A.", 1, 2)
        # 15c: the answer is exact, so no rounding instruction
        qs[15]["parts"][2]["t"] = "What was the average (mean) number of goals scored per game?"
        # 17a: clear wording
        qs[17]["parts"][0]["t"] = "Seven pupils took a spelling test marked out of 10. Their marks were 10, 3, 9, 6, 3, 9, 9. What was the average (mean) mark?"
        assert sum([10, 3, 9, 6, 3, 9, 9]) / 7 == 7
    else:
        qs[12]["parts"][0] = N("Two angles of a triangle are 64° and 39°. Find the third angle.", 77, a="77°", suf="°")
        # 13: A(-4, 1), B(2, 2), C(-2, -5): no rotations
        qs[13]["parts"][0] = C("Triangle ABC is reflected in the x-axis. Give the coordinates of the new position of point A.", -4, -1)
        qs[13]["parts"][2] = C("Triangle ABC is moved 3 squares to the right and 4 squares up. Give the coordinates of the new position of point C.", 1, -1)
    p["marks"] = sum(len(q["parts"]) for q in p["qs"])

fixA("y56-F1-A", 1); fixA("y56-F2-A", 2)

# ---------------- new Assessment B ----------------
def makeB(form, k):
    pre = f"y56{form}B-"; I = lambda n: pre + f"{n:02d}"
    qs = []
    a1, b1, c1, d1, e1, m1 = k["arith"]
    qs.append(Q(I(1), 1, "as", ["Work these out."], [N(f"Calculate {a1} + {b1}.", a1 + b1), N(f"Calculate {c1} − {d1}.", c1 - d1),
              dict(N(f"Calculate {e1} × {m1}.", e1 * m1), sk="md")]))
    fa, fb, fc, fd = k["frac"]
    qs.append(Q(I(2), 2, "frac", [], [
        N(f"What is {fa[0]}/{fa[1]} of {fa[2]}?", fa[2] * fa[0] // fa[1]),
        FR(f"Write {fb[0]}/{fb[1]} as a mixed number.", F(fb[0], fb[1]), mixed(F(fb[0], fb[1]))),
        N(f"Fill in the box: {fc[0]}/{fc[1]} = □/{fc[2]}", fc[0] * fc[2] // fc[1]),
        FR(f"Calculate {fd[0]}/{fd[1]} + {fd[2]}/{fd[3]}.", F(fd[0], fd[1]) + F(fd[2], fd[3]), mixed(F(fd[0], fd[1]) + F(fd[2], fd[3])))]))
    s1, c3, lo, hi = k["pow"]
    sq = [n * n for n in range(1, 20) if lo < n * n < hi]; assert len(sq) == 1
    qs.append(Q(I(3), 3, "pow", [], [N(f"What is {s1}²?", s1 * s1), N(f"What is {c3}³?", c3 ** 3), N(f"Which square number is between {lo} and {hi}?", sq[0])]))
    same, opts1, big = k["order"]
    qs.append(Q(I(4), 4, "order", [], [CH(f"Which of these is equal to {same[0]}?", same[1], opts1), CH("Which of these numbers is the largest?", big[0], big[1])]))
    kg, lit, rib, piece = k["units"]
    qs.append(Q(I(5), 5, "units", [], [N(f"How many grams are there in {kg} kg?", round(kg * 1000)), N(f"How many millilitres are there in {lit} litres?", round(lit * 1000)),
              N(f"A ribbon {rib} m long is cut into pieces {piece} cm long. How many pieces are there?", rib * 100 // piece)]))
    x1, x2, y1, y2 = k["dec"]
    qs.append(Q(I(6), 6, "dec", [], [N(f"Calculate {x1} + {x2}.", round(x1 + x2, 2)), N(f"Calculate {y1} − {y2}.", round(y1 - y2, 2))]))
    p1, p2, p3 = k["perc"]
    qs.append(Q(I(7), 7, "money", [], [N(f"What is {p1[0]}% of {p1[1]}?", p1[0] * p1[1] // 100),
              N(f"Write {p2[0]}/{p2[1]} as a percentage.", 100 * p2[0] // p2[1], a=f"{100 * p2[0] // p2[1]}%", suf="%"),
              N(f"A game costs £{p3[1]}. In a sale it is {p3[0]}% off. What is the sale price?", p3[1] * (100 - p3[0]) // 100, a=f"£{p3[1] * (100 - p3[0]) // 100}")]))
    (r1, r2, tot, n1, n2), (eggs, cakes, cakes2) = k["ratio"]
    qs.append(Q(I(8), 8, "ratio", [], [N(f"{tot} sweets are shared between {n1} and {n2} in the ratio {r1} : {r2}. How many sweets does {n2} get?", tot * r2 // (r1 + r2)),
              N(f"A recipe uses {eggs} eggs to make {cakes} cakes. How many eggs are needed for {cakes2} cakes?", eggs * cakes2 // cakes)]))
    (lo9, hi9, m9a, m9b), f9, (plo, phi, dsum) = k["puz"]
    both = [n for n in range(lo9, hi9) if n % m9a == 0 and n % m9b == 0]; assert len(both) == 1
    prime = [n for n in range(plo, phi) if all(n % q for q in range(2, n)) and sum(map(int, str(n))) == dsum]; assert len(prime) == 1
    qs.append(Q(I(9), 9, "puz1", [], [N(f"Which number between {lo9} and {hi9} is a multiple of both {m9a} and {m9b}?", both[0]),
              N(f"How many factors does {f9} have?", sum(1 for q in range(1, f9 + 1) if f9 % q == 0)),
              N(f"I am a prime number between {plo} and {phi}. My digits add up to {dsum}. What number am I?", prime[0])]))
    s0, st, opts10 = k["seq"]
    seq = [s0 + st * i for i in range(4)]; inseq = [o for o in opts10 if (o - s0) % st == 0 and o >= s0]; assert len(inseq) == 1
    qs.append(Q(I(10), 10, "seq", [f"This sequence goes up by {st} each time: {', '.join(map(str, seq))}, …"], [
              N("What is the next number?", s0 + 4 * st), N("What is the 10th number in the sequence?", s0 + 9 * st),
              CH("Which of these numbers is in the sequence?", str(inseq[0]), [str(o) for o in opts10])]))
    sub, mul, xin, yout = k["mach"]
    fix = F(mul * sub, mul - 1); assert fix.denominator == 1
    qs.append(Q(I(11), 11, "mach", ["A number machine works like this.", f"Number in  →  − {sub}  →  × {mul}  →  Output",
              f"For example, if you put in {sub + 2} the machine gives:  {sub + 2} → 2 → {2 * mul}."], [
              N(f"What is the output if {xin} is put in?", (xin - sub) * mul), N(f"What number was put in if the output is {yout}?", yout // mul + sub),
              N("Which number put in gives an output equal to the number put in?", int(fix))]))
    assert yout % mul == 0
    sl, rt = k["ang"]
    qs.append(Q(I(12), 12, "ang", [], [N(f"Two angles lie together on a straight line. One of them is {sl}°. What is the other?", 180 - sl, a=f"{180 - sl}°", suf="°"),
              N(f"A right-angled triangle has an angle of {rt}°. What is the size of its third angle?", 90 - rt, a=f"{90 - rt}°", suf="°")]))
    (ax, ay, s13), (mx, my) = k["coord"]
    A_, B_, C_ = (ax, ay), (ax, ay + s13), (ax + s13, ay + s13); D_ = (ax + s13, ay)
    qs.append(Q(I(13), 13, "coord", ["The diagram shows three corners, A, B and C, of a square ABCD."], [
              C("Give the coordinates of the fourth corner, D.", *D_), N("What is the area of the square in square units?", s13 * s13),
              C(f"The square is moved {abs(mx)} squares to the {'left' if mx < 0 else 'right'} and {abs(my)} squares {'down' if my < 0 else 'up'}. Give the new coordinates of C.", C_[0] + mx, C_[1] + my)],
              pic=f"cgrid A:{A_[0]},{A_[1]} B:{B_[0]},{B_[1]} C:{C_[0]},{C_[1]}"))
    r14a, r14b, (e1_, e2_) = k["round"]
    est = round(e1_, -2) * round(e2_, -1)
    qs.append(Q(I(14), 14, "round", [], [N(f"Round {r14a:,} to the nearest thousand.", round(r14a, -3)), N(f"Round {r14b} to one decimal place.", round(r14b, 1)),
              N(f"Estimate {e1_} × {e2_} by rounding {e1_} to the nearest hundred and {e2_} to the nearest ten.", int(est))]))
    title, data, step, (more1, more2), fq = k["chart"]
    tot15 = sum(v for _, v in data)
    qs.append(Q(I(15), 15, "chart", [f"The bar chart shows {title[0].lower() + title[1:]}."], [
              N("How many children are there altogether?", tot15),
              N(f"How many more children chose {more1} than {more2}?", dict(data)[more1] - dict(data)[more2]),
              FR(f"What fraction of the children chose {fq}? Give your answer in its simplest form.", F(dict(data)[fq], tot15), mixed(F(dict(data)[fq], tot15)), strict=True)],
              pic=f"chart {title};{','.join(f'{l}={v}' for l, v in data)};{step}"))
    (fs, fl), hrs, (b0, ev, after) = k["time"]
    nb = b0
    while nb <= after: nb += ev
    qs.append(Q(I(16), 16, "time", [], [T(f"A film starts at {fs // 60:02d}:{fs % 60:02d} and lasts {fl // 60} hour {fl % 60} minutes. At what time does it end?", fs + fl),
              N(f"How many minutes are there in {hrs[0]} and a half hours?", hrs[0] * 60 + 30),
              T(f"Buses leave the station every {ev} minutes, starting at {b0 // 60:02d}:{b0 % 60:02d}. What time is the first bus after {after // 60:02d}:{after % 60:02d}?", nb)]))
    lst, (cnt, mn, four) = k["mean"]
    assert sum(lst) % len(lst) == 0
    qs.append(Q(I(17), 17, "mean", [], [N(f"Find the average (mean) of {', '.join(map(str, lst[:-1]))} and {lst[-1]}.", sum(lst) // len(lst)),
              N(f"{cnt} numbers have a mean of {mn}. What is their total?", cnt * mn),
              N(f"{cnt - 1} of these numbers are {', '.join(map(str, four[:-1]))} and {four[-1]}. What is the last number?", cnt * mn - sum(four))]))
    qs.append(Q(I(18), 18, "asn", ["For each of these statements, say whether it is always true, sometimes true or never true."],
              [CH(t, v, ASN) for t, v in k["asn"]]))
    L19, W19 = k["rect"]
    side = F(2 * (L19 + W19), 4)
    qs.append(Q(I(19), 19, "rect", [f"A rectangle is {L19} cm long and {W19} cm wide."], [
              N("What is its perimeter?", 2 * (L19 + W19), a=f"{2 * (L19 + W19)} cm", suf="cm"), N("What is its area?", L19 * W19, a=f"{L19 * W19} cm²", suf="cm²"),
              N("A square has the same perimeter as the rectangle. How long is one side of the square?", float(side) if side.denominator > 1 else int(side), a=f"{float(side):g} cm", suf="cm")]))
    (nm1, nm2, times, yrs, then) = k["age"]
    # older = times * younger now; in yrs years older = 2 * younger
    young = [y for y in range(1, 30) if times * y + yrs == 2 * (y + yrs)]; assert len(young) == 1
    y0 = young[0]; o0 = times * y0
    back = [b for b in range(1, y0) if o0 - b == then * (y0 - b)]; assert len(back) == 1
    qs.append(Q(I(20), 20, "puz2", [f"{nm1} is {times} times as old as {nm2}. In {yrs} years' time {nm1} will be twice as old as {nm2}."], [
              N(f"How old is {nm2} now?", y0), N(f"How old is {nm1} now?", o0), N(f"How many years ago was {nm1} {then} times as old as {nm2}?", back[0])]))
    return {"id": f"y56-F{form}-B", "level": "y56", "form": form, "sit": "B", "marks": sum(len(q["parts"]) for q in qs), "qs": qs}
