"""Papers get harder from 1 to 60 (27 Nov). Every paper has questions 1-20 on the same topics.
For each question number, the 60 versions are sorted by a difficulty score; Paper 1 keeps its own,
Papers 2-60 get the rest from easiest to hardest. Safe to run more than once."""
import json, re, collections
def qscore(x):
    s, q = x.get('s', ''), x['q']
    nums = re.findall(r'\d+(?:\.\d+)?', q)
    big = max([len(n.split('.')[0]) for n in nums] or [1])
    return (1.0 * s.count('=') + 0.6 * len(re.split(r'\.\s+', s)) + 0.4 * big
            + 1.5 * bool(re.search(r'\d \d+/\d+', q)) + 0.8 * bool(re.search(r'\d\.\d', q))
            + len(q.split()) / 30 + 2 * (x.get('m', 1) - 1))
def gscore(g): return sum(qscore(x) for x in g) / len(g) + 0.5 * len(g)
if __name__ == '__main__':
    c = json.load(open('cem.json'))
    G = collections.defaultdict(lambda: collections.defaultdict(list))
    for x in c:
        if x.get('paper'): G[int(re.match(r'\d+', x['pq'])[0])][x['paper']].append(x)
    moved = 0
    for n, byp in G.items():
        keep = byp.pop('PT1', None)
        groups = sorted(byp.values(), key=lambda g: (round(gscore(g), 6), g[0]['q']))
        papers = sorted(byp, key=lambda p: int(p[2:]))
        for p, g in zip(papers, groups):
            for x in sorted(g, key=lambda x: x['pq']):
                part = re.sub(r'^\d+', '', x['pq'])
                if x['paper'] != p: moved += 1
                x['paper'] = p; x['id'] = f"{p}-{n:02d}{part}"; x['pq'] = f"{n}{part}"
                if 'src' in x: x['src'] = re.sub(r'practice test \d+', 'practice test ' + p[2:], x['src'])
    json.dump(c, open('cem.json', 'w'), ensure_ascii=False); print('moved', moved)
    P = collections.defaultdict(list)
    for x in c:
        if x.get('paper'): P[x['paper']].append(qscore(x))
    print([round(sum(P[f'PT{i}']) / len(P[f'PT{i}']), 1) for i in range(1, 61)])
