def lcm(a, b): return a * b // gcd(a, b)


def _pf(r, dens=(2, 3, 4, 5, 6, 8, 9, 12, 20)):
    b = r.choice(dens); a = r.choice([x for x in range(1, b) if gcd(x, b) == 1]); return F(a, b)


# 1 Written arithmetic
def s1(r, v):
    a = r.randint(11, 89) * 100 + r.randint(40, 99); b = r.randint(100 - a % 100 + 1, 98)
    while True:
        x = r.randint(2000, 9000); y = r.randint(123, 989)
        if y % 10 > x % 10 and (y // 10) % 10 > (x // 10) % 10: break
    if v % 3 == 0:
        p, q = r.randint(123, 489), r.randint(23, 67); c = (f'{p} × {q}', n(p * q), f'{p} × {q % 10} = {n(p * (q % 10))} and {p} × {q // 10 * 10} = {n(p * (q // 10 * 10))}. Add: {n(p * q)}.')
    else:
        d = r.choice([6, 7, 9, 6, 7]); q = r.randint(34, 148) if v % 3 == 1 else r.randint(152, 1100)
        c = (f'{n(d * q)} ÷ {d}', n(q), f'{d} × {n(q)} = {n(d * q)}.')
    return P('Work out:', [(f'{n(a)} + {b}', n(a + b), f'{n(a)} + {b} = {n(a + b)} (the tens carry into the hundreds).'), (f'{n(x)} − {y}', n(x - y), f'{n(x)} − {y} = {n(x - y)} (exchange in the ones and tens columns).'), c], 'Written arithmetic')


# 2 Fractions
def s2(r, v):
    while True:
        x, y = _pf(r), _pf(r)
        if x.denominator != y.denominator: break
    w1, w2 = r.randint(1, 5), r.randint(1, 4)
    A = (f'{fr(w1 + x)} + {fr(w2 + y)}', fr(w1 + w2 + x + y), f'Whole numbers: {w1 + w2}. Fractions: ' + T1._cd(x, y, '+') + f' Total: {fr(w1 + w2 + x + y)}.') if v % 2 else (f'{fr(x)} + {fr(y)}', fr(x + y), T1._cd(x, y, '+'))
    while True:
        p, q = _pf(r), _pf(r); a, b = r.randint(3, 9), r.randint(1, 4)
        if p.denominator != q.denominator and a > b and p < q: break
    X, Y = a + p, b + q; L = lcm(X.denominator, Y.denominator)
    B = (f'{fr(X)} − {fr(Y)}', fr(X - Y), f'As improper fractions over {L}: {X.numerator * L // X.denominator}/{L} − {Y.numerator * L // Y.denominator}/{L} = {(X - Y).numerator * L // (X - Y).denominator}/{L} = {fr(X - Y)}.')
    while True:
        m, k = _pf(r), _pf(r)
        if m != k: break
    C = (f'{fr(m)} × {fr(k)}', fr(m * k), f'{m.numerator * k.numerator}/{m.denominator * k.denominator} = {fr(m * k)}.') if v % 2 == 0 else (f'{fr(m)} ÷ {fr(k)}', fr(m / k), f'Turn the second fraction upside down and multiply: {fr(m)} × {inv(k)} = {fr(m / k)}.')
    t = v % 3
    if t == 0: u = r.randint(2, 8); mm = r.randint(4, 20); D_ = (f'Fill in the box:  □ ÷ 1/{u} = {mm}', fr(F(mm, u)), f'Dividing by 1/{u} multiplies by {u}, so □ × {u} = {mm} and □ = {fr(F(mm, u))}.')
    elif t == 1: k2 = r.choice([3, 4, 5, 6, 7]); val = k2 * (r.randint(0, 2) + r.choice([F(1, 3), F(2, 3), F(1, 4), F(3, 4), F(2, 5), F(1, 6), F(3, 8)])); D_ = (f'Fill in the box:  {k2} × □ = {fr(val)}', fr(val / k2), f'□ = {fr(val)} ÷ {k2} = {val.numerator}/{val.denominator * k2} = {fr(val / k2)}.')
    else: g, h = _pf(r), _pf(r); D_ = (f'Fill in the box:  □ − {fr(g)} = {fr(h)}', fr(g + h), f'□ = {fr(h)} + {fr(g)} = {fr(g + h)}.')
    return P('Work out the following. Give each answer in its simplest form.', [A, B, C, D_], 'Fractions')


# 3 Powers
def s3(r, v):
    a, k = r.choice([(3, 5), (2, 6), (2, 7), (5, 3), (5, 4), (4, 4), (3, 4), (2, 8), (6, 3), (10, 5), (4, 3), (3, 6), (7, 3), (2, 9)])
    A = (f'Fill in the box:  {a}^□ = {n(a ** k)}', k, ', '.join(f'{sup(a, i)} = {n(a ** i)}' for i in range(1, k + 1)) + '.')
    t = v % 4
    if t == 0: p, q = r.randint(5, 9), r.randint(2, 4); B = (f'Fill in the box:  2^□ ÷ {2 ** q} = {2 ** (p - q)}', p, f'{2 ** (p - q)} × {2 ** q} = {2 ** p} = {sup(2, p)}.')
    elif t == 1: x, y = r.choice([(3, 4), (2, 5), (2, 6), (3, 5), (4, 5), (2, 7)]); B = (f'Which is larger, {sup(x, y)} or {sup(y, x)}, and by how much?', f'{sup(x, y) if x ** y > y ** x else sup(y, x)}, by {abs(x ** y - y ** x)}', f'{sup(x, y)} = {x ** y} and {sup(y, x)} = {y ** x}.')
    elif t == 2: x, y, z = r.choice([(3, 4, 5), (5, 12, 13), (6, 8, 10), (8, 15, 17), (9, 12, 15)]); B = (f'Fill in the box:  {x}² + {y}² = □²', z, f'{x * x} + {y * y} = {z * z} = {z}².')
    else: x, y = r.choice([(2, 3), (3, 2), (2, 4), (5, 2), (4, 2)]); c, d = r.choice([(3, 2), (5, 2), (10, 2), (2, 3)]); B = (f'Work out {sup(x, y)} × {sup(c, d)}', n(x ** y * c ** d), f'{sup(x, y)} = {x ** y} and {sup(c, d)} = {c ** d}. {x ** y} × {c ** d} = {n(x ** y * c ** d)}.')
    b0 = r.choice([4, 6, 7, 9])
    return P(f'{sup(b0, 5)} means {" × ".join([str(b0)] * 5)}.', [A, B], 'Powers')


# 4 Ordering
def s4(r, v):
    target, ts = r.choice([(F(1, 2), '1/2'), (F(3, 4), '3/4'), (F(1, 4), '1/4'), (F(2, 3), '2/3')])
    while True:
        d = {}
        while len(d) < 5:
            if r.random() < .55:
                b = r.choice([3, 4, 5, 6, 8, 9, 12, 20]); a = r.choice([x for x in range(1, b) if gcd(x, b) == 1]); val = F(a, b); s = f'{a}/{b}'
            else: val = F(r.randint(15, 92), 100) if r.random() < .7 else F(r.randint(150, 920), 1000); s = dec(D(val.numerator) / D(val.denominator))
            if val not in d: d[val] = s
        ds = sorted(abs(x - target) for x in d)
        if target not in d and ds[0] != ds[1] and sum('/' in s for s in d.values()) in (2, 3): break
    items = list(d.items()); r.shuffle(items); best = min(d, key=lambda x: abs(x - target))
    why = 'As decimals: ' + ', '.join(f'{s} = {float(x):.3g}' for x, s in sorted(items)) + '.'
    return P(['Here are five numbers:', '      '.join(s for _, s in items)], [('Which is the smallest?', d[min(d)], why), ('Which is the largest?', d[max(d)], why), (f'Which is closest to {ts}?', d[best], why + f' {ts} = {float(target):.3g}.')], 'Ordering fractions and decimals')


# 5 Decimals

    d = D(r.choice(['0.2', '0.3', '0.4', '0.6', '0.7', '0.8', '0.05', '0.04', '1.2'])); q = r.randint(3, 40); k = 10 ** (-d.as_tuple().exponent)

    C = (f'Fill in the box:  □ × {n(m)} = {dec(z)}', dec(z / m), f'{dec(z)} ÷ {n(m)} = {dec(z / m)}.') if v % 2 else (f'Fill in the box:  {dec(z)} ÷ □ = {dec(z / m)}', n(m), f'The digits have moved {len(str(m)) - 1} place{"s" if m > 10 else ""} to the right, so □ = {n(m)}.')
    return P('Work out the following. Give your answers as decimals.', [(f'{dec(x)} × {dec(y)}', dec(x * y), f'{int(x * 10)} × {int(y.scaleb(-y.as_tuple().exponent))} = {int(x * 10) * int(y.scaleb(-y.as_tuple().exponent))}; the question has {1 - y.as_tuple().exponent} decimal places, so {dec(x * y)}.'),
                                                                   (f'{dec(d * q)} ÷ {dec(d)}', q, f'Multiply both by {k}: {dec(d * q * k)} ÷ {dec(d * k)} = {q}.'), C], 'Decimals')

# 8 Ratio
def s8(r, v):
    t = v % 4

# 9 Puzzle (rotates)
def s9(r, v):
    t = v % 5
    if t == 0:
        k = r.randint(4, 7); m = r.randint(3, 6); ds = sorted(r.sample(range(1, 10), k)); ls = 'ABCDEF'[:m]
        return P(f'A code is made using the digits {", ".join(map(str, ds[:-1]))} and {ds[-1]}.', [('How many different 3-digit codes are possible if digits can be repeated?', k ** 3, f'{k} × {k} × {k} = {k ** 3}.'), ('How many different 3-digit codes are possible if no digit can be used more than once?', k * (k - 1) * (k - 2), f'{k} × {k - 1} × {k - 2} = {k * (k - 1) * (k - 2)}.'),
                                                                                                  (f'A different code is one letter from {", ".join(ls[:-1])} or {ls[-1]} followed by two different digits from the list above. How many of these codes are possible?', m * k * (k - 1), f'{m} × {k} × {k - 1} = {m * k * (k - 1)}.')], 'Counting puzzle')
    if t == 1:
        while True:
            x, y = r.randint(213, 687), r.randint(124, 298)
            if x % 10 + y % 10 >= 10 and '0' not in str(x) + str(y): break
        xs, ys = str(x), str(y)
        while True:
            p, q = r.randint(512, 987), r.randint(134, 478)
            if p % 10 < q % 10 and '0' not in str(p) + str(q): break
        ps, qs = str(p), str(q); k = r.choice([4, 5, 6, 7]); c = r.randint(11, 29); box = r.randint(3, 9)
        return P('In each calculation a box □ hides one digit.', [(f'{xs[0]}□{xs[2]} + {ys[0]}{ys[1]}□ = {x + y}.  Write the two missing digits in order.', f'{xs[1]} and {ys[2]}', f'Ones: {xs[2]} + {ys[2]} = {int(xs[2]) + int(ys[2])}, carry 1. Then the tens need {xs[1]}. Check: {x} + {y} = {x + y}.'),
                                                                 (f'{ps[0]}{ps[1]}□ − {qs[0]}□{qs[2]} = {p - q}.  Write the two missing digits in order.', f'{ps[2]} and {qs[1]}', f'Check: {p} − {q} = {p - q}.'),
                                                                 (f'{k} × (□ + {c}) = {k * (box + c)}.  What is the missing digit?', box, f'{k * (box + c)} ÷ {k} = {box + c}, and {box + c} − {c} = {box}.')], 'Missing-digit puzzle')
    if t == 2:
        p, q = 5 * r.randint(3, 12), 5 * r.randint(4, 16); a = r.randint(2, 5); b1 = r.randint(1, 3); b2 = b1 + 2; x, y = r.choice([('pencils', 'pens'), ('apples', 'pears'), ('buns', 'cakes')])
        c1, c2 = a * p + b2 * q, a * p + b1 * q
        return P(f'{a} {x} and {b2} {y} cost {money(c1)}. {a} {x} and {b1} {y} cost {money(c2)}.', [(f'What do 2 {y} cost?', money(2 * q), f'The only difference is 2 {y}: {money(c1)} − {money(c2)} = {money(2 * q)}.'), (f'What does 1 {y[:-1]} cost?', pm(q), f'{money(2 * q)} ÷ 2 = {pm(q)}.'), (f'What does 1 {x[:-1]} cost?', pm(p), f'{a} {x} cost {money(c2)} − {money(b1 * q)} = {money(a * p)}. {money(a * p)} ÷ {a} = {pm(p)}.')], 'Price puzzle')
    if t == 3:
        m, w, c = r.randint(1200, 4000), r.randint(1200, 4000), r.randint(1500, 5000); T = m + w + c; town = r.choice(['Grangeford', 'Little Marsham', 'Northby', 'Oakhampton', 'Westbrook'])

The men and women together number {n(m + w)}.', [('How many women are there?', n(w), f'{n(T)} − {n(m + c)} = {n(w)}.'), ('How many children are there?', n(c), f'{n(T)} − {n(m + w)} = {n(c)}.'), ('How many men are there?', n(m), f'{n(T)} − {n(w)} − {n(c)} = {n(m)}.')], 'Three-group puzzle')
    while True:

        if len({a + b for a, b in combinations(ps, 2)}) == 6 and sum(ps) % 4 == 0: break

# 18 Always, sometimes, never
def s18(r, v):
    cat = {k: [x for x in ASN if x[1] == k] for k in ('Always', 'Sometimes', 'Never')}
    for k in cat: random.Random(99).shuffle(cat[k])
    pat = [['Always', 'Sometimes', 'Never', 'Sometimes'], ['Always', 'Always', 'Sometimes', 'Never'], ['Sometimes', 'Never', 'Always', 'Never']][v % 3]
    cnt = {k: sum(p.count(k) * ((v - j) // 3 + (1 if j < v % 3 else 0)) for j, p in enumerate([['Always', 'Sometimes', 'Never', 'Sometimes'], ['Always', 'Always', 'Sometimes', 'Never'], ['Sometimes', 'Never', 'Always', 'Never']])) for k in cat}
    pick = []
    for k in pat:
        pick.append(cat[k][cnt[k] % len(cat[k])]); cnt[k] += 1
    r.shuffle(pick)
    return P('For each statement, write whether it is ALWAYS true, SOMETIMES true or NEVER true.', [(s, a, why) for s, a, why in pick], 'Always, sometimes, never')

def make_paper(i):
    qs = []
    for k, f in enumerate(SLOTS):
        r = random.Random(7000 + 131 * i + k)
        v = i + (k * 3 if k in (8, 19, 14) else k)   # stagger the rotating slots so papers differ
