"""Make fresh questions for one topic on demand.
   python3 ondemand.py            -> list the 30 topic codes and their question types
   python3 ondemand.py RAT 20 7   -> 20 new ratio questions, seed 7 (change the seed for a different set)
   python3 ondemand.py TIM 10 3 "Time zones"  -> only one question type"""
import sys, json, random
from topics1 import TOPICS1
from topics2 import TOPICS2
from topics3 import TOPICS3
ALL = {c: (t, f) for c, t, f in TOPICS1 + TOPICS2 + TOPICS3}
if len(sys.argv) < 2:
    for c, (t, fs) in ALL.items():
        r = random.Random(1); print(c, '-', t, '::', ', '.join(dict.fromkeys(f(r)['type'] for f in fs for _ in range(6))))
    sys.exit()
code, n, seed = sys.argv[1], int(sys.argv[2]) if len(sys.argv) > 2 else 10, int(sys.argv[3]) if len(sys.argv) > 3 else 1
only = sys.argv[4] if len(sys.argv) > 4 else None
title, fs = ALL[code]; r = random.Random(seed * 7919 + 13); out, seen, i, tries = [], set(), 0, 0
while len(out) < n and tries < 5000:
    q = fs[i % len(fs)](r); i += 1; tries += 1
    if (only and q['type'] != only) or q['text'] in seen: continue
    seen.add(q['text']); out.append({k: q[k] for k in ('text', 'ans', 'sol', 'type', 'table', 'alt') if q.get(k)})
print(json.dumps(out, ensure_ascii=False, indent=1))
