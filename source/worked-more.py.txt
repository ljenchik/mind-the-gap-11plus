"""Fuller worked answers (27 Nov, second pass). Safe to run more than once."""
import json, re
from fractions import Fraction as F
def num(t):
    t = t.replace('−', '-').replace('£', '').replace(',', '').strip()
    m = re.fullmatch(r'-?\d+(?:\.\d+)?', t)
    return float(t) if m else None
def dec(fr):
    v = fr.numerator / fr.denominator
    s = f"{v:.3f}".rstrip('0').rstrip('.')
    return s if F(s) == fr else s  # recurring ones shown to 3 places
PLACE = {'10': 'ones', '100': 'tens', '1000': 'hundreds', 'thousand': 'hundreds', 'hundred': 'tens', 'ten': 'ones'}
VUL = {'½': '1/2', '¼': '1/4', '¾': '3/4', '⅓': '1/3', '⅔': '2/3', '⅛': '1/8'}
def vulgar(t):
    t = re.sub(r'(\d)([½¼¾⅓⅔⅛])', lambda m: m[1] + ' ' + VUL[m[2]], t)
    return re.sub(r'[½¼¾⅓⅔⅛]', lambda m: VUL[m[0]], t)
def fix(x):
    for f in ('q', 's', 'a', 'hint'):
        if f in x and re.search(r'[½¼¾⅓⅔⅛]', x[f]): x[f] = vulgar(x[f])
    q, s, a = x['q'], x.get('s', ''), x['a']; n = 0
    # 1. drop "So the answer is X." when the step before already ends with the same number
    m = re.search(r'^(.*?(-?£?[\d.]+)[^\d]{0,12})\. So the answer is ([^.]+(?:\.\d+)?[^.]*)\.$', s)
    if m and re.fullmatch(r'−?\d+(?:\.\d+)?', a) and num(m[2]) is not None and num(m[3]) == num(m[2]):
        s = m[1] + '.'; n += 1
    # 2. fractions and decimals in a list: write every fraction as a decimal, then put them in order
    last = q.split('\n')
    m = re.search(r'(?:numbers|these):?\s+((?:\d+/\d+|\d*\.\d+|\d+)(?:,?\s+(?:and\s+)?(?:\d+/\d+|\d*\.\d+|\d+))+)', q)
    if m and re.search(r'smallest|largest', q.split('\n')[-1]) and len(s) < 70 and '/' in m[1] and '.' in m[1]:
        items = re.findall(r'\d+/\d+|\d*\.\d+|\d+', m[1])
        val = {t: (F(t) if '/' in t else F(t)) for t in items}
        fr = [f"{t} = {dec(val[t])}" for t in items if '/' in t]
        order = sorted(items, key=lambda t: val[t])
        big = 'largest' in q.split('\n')[-1]
        pick = order[-1] if big else order[0]
        if pick == a:
            ds = [dec(val[t]) if '/' in t else t for t in order]
            s = (f"Write each fraction as a decimal: {', '.join(fr)}. In order, smallest first: {', '.join(ds)}. "
                 f"The {'largest' if big else 'smallest'} is {dec(val[pick])}" + (f", which is {pick}." if '/' in pick else '.'))
            n += 1
    # 3. rounding with only "Look at the ... digit"
    m = re.fullmatch(r'Look at the (\w+) digit: (\d+)\.', s)
    r = re.search(r'Round (\d+) to the nearest (ten|hundred|thousand|10|100|1000)\b', q)
    if m and r:
        v = r[1]; step = {'ten': 10, 'hundred': 100, 'thousand': 1000}.get(r[2], None) or int(r[2])
        d = (int(v) // (step // 10)) % 10
        s = f"Look at the {m[1]} digit. It is {d}, so round {'up' if d >= 5 else 'down'}. {v} rounds to {m[2]}."; n += 1
    # 4. "10% of A = B, so P% = C" : show the multiplying
    def pc(mm):
        A, B, P, C = mm[1], mm[2], int(mm[3]), mm[4]
        k = F(P, 10)
        if k.denominator == 1 and k > 1 and num(B) is not None and abs(num(B) * k - num(C)) < 1e-9:
            return f"10% of {A} = {B}, so {P}% = {k} × {B} = {C}"
        return mm[0]
    s2 = re.sub(r'10% of (£?\d+(?:\.\d+)?) = (£?\d+(?:\.\d+)?), so (\d+)% = (£?\d+(?:\.\d+)?)(?! =)', pc, s)
    if s2 != s: s = s2; n += 1
    # money with one decimal place: £6.5 -> £6.50
    s = re.sub(r'£(\d+)\.(\d)(?![\d])', r'£\1.\g<2>0', s)
    # 5. list answers use " · " inside the app: write them with commas in sentences
    s2 = re.sub(r'(So the answer is [^.]*?) · ', lambda mm: mm[1] + ', ', s)
    while s2 != s: s, s2 = s2, re.sub(r'(So the answer is [^.]*?) · ', lambda mm: mm[1] + ', ', s2)
    if s != x.get('s', ''): x['s'] = s; return 1
    return 0
if __name__ == '__main__':
    for f in ('questions.json', 'cem.json'):
        d = json.load(open(f)); k = sum(fix(x) for x in d); json.dump(d, open(f, 'w'), ensure_ascii=False); print(f, k)
