import json
t = open('template.html').read()
d = lambda f: json.dumps(json.load(open(f)), ensure_ascii=False)
out = t.replace('__DATA__', d('questions.json')).replace('__HELP__', d('help.json')).replace('__STARTER__', d('starter-data.json'))
open('index.html', 'w').write(out)
print('built', len(out))
