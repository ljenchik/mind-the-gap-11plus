def text_of(stem, q, lab): return q if stem in PLAIN else f"{stem}\n{lab}) {q}"

def patch_reviewed(qs):
    """Review fixes to papers 11 and 12 (23 Nov), keeping everything else as reviewed."""
    by = {q["id"]: q for q in qs}
    def put(i, t, a, s): by[i].update(q=t, a=a, s=s); by[i].pop("wrong", None)
    q = by["PT11-11c"]; stem = q["q"].split("\nc)")[0]
    put("PT11-11c", stem + "\nc) Ali also has 4 jumps, with a mean of 160 cm. How much further did Ali jump altogether than Sam?", "24 cm",
        "Each mean is a total shared out over 4 jumps. The means differ by 6 cm, so the totals differ by 4 × 6 = 24 cm.")
    put("PT11-12c", "Three angles meet at a point. Two of them are 110° and 95°. What is the third?", "155°", "Angles around a point add up to 360°. 360° − 110° − 95° = 155°.")
    q = by["PT11-17c"]; stem = q["q"].split("\nc)")[0]
    put("PT11-17c", stem + "\nc) In how many of the 36 results is the red number bigger than the blue number?", "15",
        "Red 2: blue 1 (1 way). Red 3: 2 ways. Red 4: 3, red 5: 4, red 6: 5. 1 + 2 + 3 + 4 + 5 = 15.")
    # review 23 Nov (2): questions 4, 5, 6 and 9 were too alike in papers 11 and 12 – a different kind of task in each
    fresh = {("PT11", 4): (V.ord_jars, "order"), ("PT11", 5): (V.m_kitchen, "units"), ("PT11", 6): (V.p_test, "money"), ("PT11", 9): (V.t_swim, "time"),
             ("PT12", 4): (V.ord_race, "order"), ("PT12", 5): (V.m_pool, "units"), ("PT12", 6): (V.p_school, "money"), ("PT12", 9): (V.t_bus, "time")}
    for (pp, n), (gen, sk) in fresh.items():
        stem, parts = gen(random.Random(f"{pp}-{n}-v3"))[:2]
        for k, p in enumerate(parts):
            q = by[f"{pp}-{n:02d}{'abc'[k]}"]
            q.update(q=text_of(stem, p[0], "abc"[k]), a=p[1], s=p[2], skill=sk, k=SKILLS[sk], t=SKILLS[sk], tab=SKILLS[sk], bp=BLUE[sk]); q.pop("wrong", None)
            if len(p) > 3 and p[3]: q["wrong"] = [w for w in p[3] if w != p[1]][:3]
    put("PT12-12c", "A right angle is split into two angles. One of them is 34°. What is the other?", "56°", "A right angle is 90°. 90° − 34° = 56°.")
    assert 4 * 160 - 4 * 154 == 24 and 360 - 110 - 95 == 155

# 23 Nov: papers 11 and 12 re-ordered to the real paper order; new slot -> old question number (others are new questions)
MOVE = {"PT11": {1: 1, 2: 3, 4: 4, 5: 5, 6: 6, 7: 7, 8: 19, 10: 12, 12: 11, 13: 9, 15: 14, 16: 16, 17: 15, 18: 10, 19: 17, 20: 20},
        "PT12": {1: 1, 2: 3, 4: 4, 5: 5, 6: 6, 7: 7, 8: 19, 10: 12, 12: 11, 13: 9, 15: 14, 16: 16, 17: 15, 18: 10, 20: 20}}

def main():
    bank = json.load(open("/home/claude/live3/source/questions.json"))
    old = json.load(open("/home/claude/data/pt11_30.json"))
    keep = [q for q in old if q["paper"] in ("PT11", "PT12")]; patch_reviewed(keep)
    newp = lambda q: str(q.get("paper", "")).startswith("PT") and int(q["paper"][2:]) > 12
    key = lambda q: q["q"] + "|" + q.get("pic", "")
    seen = {key(q) for q in bank if not str(q.get("paper", "")).startswith("PT") or int(q["paper"][2:]) not in range(11, 31)} | {key(q) for q in keep}
    R = random.Random("papers-11-30-v3"); out = []; mix = []
    old_by = {q["id"]: q for q in keep}
    for i, pn in enumerate(range(11, 31)):
        code = f"PT{pn}"; used = set(); kinds = []
        for j, pool in enumerate(SLOTS):
            if j + 1 in MOVE.get(code, {}):          # papers 11 and 12: reviewed question moved to its real-paper position
                on = MOVE[code][j + 1]
                for k in "abc":
                    d = dict(old_by[f"{code}-{on:02d}{k}"]); d.update(id=f"{code}-{j + 1:02d}{k}", pq=f"{j + 1}{k}"); out.append(d)
                kinds.append(f"(old {on})"); continue
            order = [pool[(i + j + k) % len(pool)] for k in range(len(pool))]
            done = False
            for name, gen, sk in order:
                if name in used: continue
                for _ in range(300):
                    res = gen(R); stem, parts = res[0], res[1]; pic = res[2] if len(res) > 2 else None
                    if len(parts) != 3 or any(not p[1] or "-" in p[1] for p in parts): continue
                    texts = [text_of(stem, p[0], "abc"[k]) for k, p in enumerate(parts)]
                    keys = [t + "|" + (pic or "") for t in texts]
                    if keys[0] in seen: continue
                    seen.update(keys); done = True; break
                if done: break
            assert done, f"no question for {code} slot {j + 1}"
            used.add(name); kinds.append(name); qn = j + 1
            for k, p in enumerate(parts):
                d = dict(id=f"{code}-{qn:02d}{'abc'[k]}", q=texts[k], a=p[1], s=p[2], k=SKILLS[sk], t=SKILLS[sk], tab=SKILLS[sk], m=1, wp=False, lv="Challenge", yr=6,
                         skill=sk, src=f"Original CSSE-style practice test {pn}, written for this app", bp=BLUE[sk], paper=code, pq=f"{qn}{'abc'[k]}")
                if pic: d["pic"] = pic
                if len(p) > 3 and p[3]: d["wrong"] = [w for w in p[3] if w != p[1]][:3]
                out.append(d)
        mix.append((code, kinds))
    json.dump(out, open("/home/claude/data/pt11_30.json", "w"), ensure_ascii=False)
    for code, kinds in mix: print(code, " ".join(kinds))
    print(len(out), "questions")

if __name__ == "__main__":
    main()
