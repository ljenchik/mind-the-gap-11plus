def primes(a, b): return [n for n in range(max(a, 2), b + 1) if all(n % d for d in range(2, int(n ** .5) + 1))]
def factors(n): return [d for d in range(1, n + 1) if n % d == 0]
def fr(x): x = F(x); return f"{x.numerator}/{x.denominator}"

# ---------- Factors and multiples ----------
def fact(R):
    n = R.choice([12, 18, 20, 24, 28, 30, 32, 36, 40, 42, 45, 48, 50, 54, 56, 60, 63, 64, 72, 80, 84, 90, 96, 100])
    a, b = R.sample([4, 6, 8, 9, 10, 12, 15], 2); l = a * b // __import__("math").gcd(a, b)
    x, y = R.sample([12, 16, 18, 20, 24, 27, 30, 36, 40, 42, 45, 48], 2); h = __import__("math").gcd(x, y)
    k = R.random()
    if k < .5:
        p3 = (f"Buses leave every {a} minutes and trams every {b} minutes. Both leave at 9:00. After how many minutes do they next leave together?", f"{l} minutes",
              f"Find the lowest common multiple of {a} and {b}: {l}. So {l} minutes.")
    else:
        p3 = (f"What is the highest common factor of {x} and {y}?", str(h), f"Factors of {x}: {', '.join(map(str, factors(x)))}. Factors of {y}: {', '.join(map(str, factors(y)))}. The highest in both is {h}.")
    fs = factors(n)
    return (f"Factors and multiples.", [
        (f"How many factors does {n} have?", str(len(fs)), f"The factors of {n} are {', '.join(map(str, fs))}. That is {len(fs)}."),
        (f"What is the smallest number that is a multiple of both {a} and {b}?", str(l), f"Multiples of {max(a, b)}: {', '.join(str(max(a, b) * i) for i in range(1, l // max(a, b) + 1))}. The first that {min(a, b)} also divides is {l}."),
        p3])

# ---------- Prime numbers ----------
def prime(R):
    lo = R.choice([20, 30, 40, 50, 60, 70, 80]); hi = lo + 10; ps = primes(lo, hi)
    if not ps: return prime(R)
    n = R.choice([12, 18, 20, 24, 28, 30, 36, 40, 42, 44, 45, 48, 50, 52, 54, 60, 63, 66, 70, 72, 75, 84, 90, 98])
    pf = []; m = n; d = 2
    while m > 1:
        while m % d == 0: pf.append(d); m //= d
        d += 1
    s = R.choice([n for n in range(10, 41, 2)]); pairs = [(p, s - p) for p in primes(2, s // 2) if (s - p) in primes(2, s)]
    k = R.random()
    if k < .5:
        p3 = (f"In how many ways can {s} be written as the sum of two prime numbers? (3 + 5 and 5 + 3 count as the same way.)", str(len(pairs)),
              f"{', '.join(f'{a} + {b}' for a, b in pairs)}. That is {len(pairs)} way{'s' if len(pairs) != 1 else ''}.")
    else:
        big = max(pf)
        p3 = (f"What is the largest prime factor of {n}?", str(big), f"{n} = {' × '.join(map(str, pf))}. The largest prime factor is {big}.")
    kind = R.random(); start = R.randint(20, 110); nxt = primes(start + 1, start + 40)[0]
    cands = R.sample(range(21, 120, 2), 6); pr = [c for c in cands if c in primes(2, 120)]
    if kind < .34: p1 = (f"How many prime numbers are there between {lo} and {hi}?", str(len(ps)), f"Check each number for factors: the primes are {', '.join(map(str, ps))}. That is {len(ps)}.")
    elif kind < .67: p1 = (f"What is the first prime number after {start}?", str(nxt), f"Check {start + 1}, {start + 2}, … for factors. The first with no factors except 1 and itself is {nxt}.")
    else:
        if len(pr) != 1: return prime(R)
        p1 = ("Which of these numbers is prime?", str(pr[0]), f"Each of the others has a factor other than 1 and itself; {pr[0]} does not.", [str(c) for c in cands if c != pr[0]][:3])
    return ("Prime numbers.", [
        p1,
        (f"Write {n} as a product of prime numbers. What do the prime factors add up to? (Count each one as many times as it appears.)", str(sum(pf)),
         f"{n} = {' × '.join(map(str, pf))}. {' + '.join(map(str, pf))} = {sum(pf)}."),
        p3])

# ---------- × and ÷ by 10, 100, 1000 ----------
def place(R):
    def dz(x): return format(x, "f").rstrip("0").rstrip(".") if "." in format(x, "f") else format(x, "f")
    from decimal import Decimal as D
    a = D(R.randint(12, 999)) / D(R.choice([10, 100])); m = R.choice([10, 100, 1000])
    b = D(R.randint(30, 9900)); dv = R.choice([10, 100, 1000])
    c = D(R.randint(11, 99)) / 10; tgt = c * R.choice([100, 1000]); mul = int(tgt / c)
    n = R.randint(10000, 99999); pos = R.choice([(1000, "thousands"), (100, "hundreds"), (10, "tens")])
    digit = n // pos[0] % 10
    k = R.random()
    p3 = (f"Fill in the box: {dz(c)} × □ = {dz(tgt)}", str(mul), f"{dz(tgt)} ÷ {dz(c)} = {mul}.") if k < .5 else \
         (f"What is the value of the digit {digit} in {n:,}?" if str(digit) in str(n) and str(n).count(str(digit)) == 1 else f"What is {n:,} ÷ 100?",
          str(digit * pos[0]) if str(n).count(str(digit)) == 1 else dz(D(n) / 100),
          f"The {digit} is in the {pos[1]} place, so it is worth {digit * pos[0]}." if str(n).count(str(digit)) == 1 else f"Move each digit two places to the right: {dz(D(n) / 100)}.")
    return ("× and ÷ by 10, 100, 1000.", [
        (f"Calculate {dz(a)} × {m}.", dz(a * m), f"Move each digit {len(str(m)) - 1} place{'s' if m > 10 else ''} to the left: {dz(a * m)}."),
        (f"Calculate {b} ÷ {dv}.", dz(b / dv), f"Move each digit {len(str(dv)) - 1} place{'s' if dv > 10 else ''} to the right: {dz(b / dv)}."),
        p3])

# ---------- Fractions, decimals and percentages ----------
FDP = [(F(1, 2), "0.5", 50), (F(1, 4), "0.25", 25), (F(3, 4), "0.75", 75), (F(1, 5), "0.2", 20), (F(2, 5), "0.4", 40), (F(3, 5), "0.6", 60), (F(4, 5), "0.8", 80),
       (F(1, 10), "0.1", 10), (F(3, 10), "0.3", 30), (F(7, 10), "0.7", 70), (F(9, 10), "0.9", 90), (F(1, 20), "0.05", 5), (F(3, 20), "0.15", 15), (F(7, 20), "0.35", 35),
       (F(9, 20), "0.45", 45), (F(1, 25), "0.04", 4), (F(6, 25), "0.24", 24), (F(1, 8), "0.125", 12.5), (F(3, 8), "0.375", 37.5), (F(11, 25), "0.44", 44), (F(13, 20), "0.65", 65)]
def fdp(R):
    (f1, d1, p1), (f2, d2, p2), (f3, d3, p3_) = R.sample(FDP, 3)
    opts = [f"{fr(f3)}", d3, f"{p3_:g}%"]; R.shuffle(opts)
    odd = R.choice([0, 1, 2]); wrongv = {0: f"{fr(f3 + F(1, 10))}", 1: str(float(d3) + 0.05)[:5], 2: f"{p3_ + 5:g}%"}[odd]
    items = [fr(f3), d3, f"{p3_:g}%"]; items[odd] = wrongv
    return ("Fractions, decimals and percentages.", [
        (f"Write {fr(f1)} as a decimal.", d1, f"{fr(f1)} = {f1.numerator} ÷ {f1.denominator} = {d1}."),
        (f"Write {d2} as a percentage.", f"{p2:g}%", f"{d2} = {p2:g} hundredths = {p2:g}%."),
        ("Which one of these is NOT equal to the other two?", items[odd], f"{fr(f3)} = {d3} = {p3_:g}%. The odd one out is {items[odd]}.", [items[i] for i in range(3) if i != odd])])

# ---------- Money ----------
def cash(R):
    return R.choice([C.FAMS['shop'], C.FAMS['bake'], cash_change, cash_tickets])(R)
def cash_change(R):
    items = R.sample([("a sandwich", 285), ("a drink", 95), ("an apple", 45), ("a cookie", 120), ("crisps", 65), ("a yoghurt", 80)], 3)
    paid = R.choice([500, 1000]); tot = sum(p for _, p in items)
    if tot >= paid: return cash_change(R)
    n = R.randint(3, 6)
    return (f"In the school café, {', '.join(f'{n_} costs {money(p)}' for n_, p in items)}.", [
        (f"How much do {items[0][0]}, {items[1][0]} and {items[2][0]} cost altogether?", money(tot), f"{' + '.join(money(p) for _, p in items)} = {money(tot)}."),
        (f"How much change is there from {money(paid)}?", money(paid - tot), f"{money(paid)} − {money(tot)} = {money(paid - tot)}."),
        (f"Every day for {n} days, Ella buys {items[1][0]}. How much does she spend?", money(n * items[1][1]), f"{n} × {money(items[1][1])} = {money(n * items[1][1])}.")])
def cash_tickets(R):
    ad, ch = R.choice([1250, 1500, 1800, 2100]), R.choice([650, 750, 900, 1050]); a, c = R.randint(1, 3), R.randint(2, 4)
    fam = R.choice([4000, 4500, 5000]); off = R.choice([10, 20, 25])
    norm = 2 * ad + 2 * ch
    if norm <= fam: return cash_tickets(R)
    return (f"A theme park charges {money(ad)} for an adult and {money(ch)} for a child.", [
        (f"How much do {a} adult{'s' if a > 1 else ''} and {c} children pay?", money(a * ad + c * ch), f"{a} × {money(ad)} + {c} × {money(ch)} = {money(a * ad + c * ch)}."),
        (f"A family ticket for 2 adults and 2 children costs {money(fam)}. How much does it save?", money(norm - fam) if norm > fam else money(0),
         f"Separate tickets: 2 × {money(ad)} + 2 × {money(ch)} = {money(norm)}. {money(norm)} − {money(fam)} = {money(norm - fam)}."),
        (f"On Mondays every ticket is {off}% off. How much is one adult ticket on a Monday?", money(ad * (100 - off) // 100), f"{off}% of {money(ad)} is {money(ad * off // 100)}. {money(ad)} − {money(ad * off // 100)} = {money(ad * (100 - off) // 100)}.")])

# ---------- Recipes, maps and rates ----------
def rates(R):
    return R.choice([RF.recipe, RF.scale, rates_speed, rates_workers])(R)
def rates_speed(R):
    v = R.choice([30, 40, 45, 50, 60, 72]); h = R.choice([2, 3, 4]); d = v * R.choice([1, 2, 3]) // 2 * 2; mins = d * 60 // v
    if d * 60 % v: return rates_speed(R)
    return (f"A car travels at a steady {v} km per hour.", [
        (f"How far does it travel in {h} hours?", f"{v * h} km", f"{v} × {h} = {v * h} km."),
        (f"How many minutes does it take to travel {d} km?", f"{mins} minutes", f"In 1 hour (60 minutes) it goes {v} km. {d} km takes {d} ÷ {v} × 60 = {mins} minutes."),
        (f"A bike goes {v // 3 if v % 3 == 0 else v // 2} km in an hour. How many times as fast is the car?", str(3 if v % 3 == 0 else 2), f"{v} ÷ {v // 3 if v % 3 == 0 else v // 2} = {3 if v % 3 == 0 else 2}.")])
def rates_workers(R):
    w, d = R.choice([(2, 12), (3, 8), (4, 6), (6, 4), (4, 9), (3, 12)]); total = w * d; w2 = R.choice([x for x in (2, 3, 4, 6, 8, 12) if total % x == 0 and x != w])
    pages, mins = R.choice([(3, 4), (5, 6), (4, 5), (6, 10)]); t = R.choice([30, 60, 90, 120])
    if t * pages % mins: return rates_workers(R)
    return ("Working at a steady rate.", [
        (f"{w} painters take {d} days to paint a school. How long would 1 painter take?", f"{total} days", f"1 painter does the work of {w} on his own: {w} × {d} = {total} days."),
        (f"How long would {w2} painters take?", f"{total // w2} days", f"{total} ÷ {w2} = {total // w2} days."),
        (f"A printer prints {pages} pages every {mins} seconds. How many pages does it print in {t} seconds?", str(t * pages // mins), f"{t} ÷ {mins} = {t / mins:g} lots of {mins} seconds. {t / mins:g} × {pages} = {t * pages // mins}.")])

def puz1(R):

def digits_puzzle(R):
    a, b = R.randint(12, 49), R.randint(12, 49); s = a + b
    ds = sorted(R.sample(range(1, 10), 4)); big = int("".join(map(str, sorted(ds, reverse=True)[:2]))) * 0
    hi = int("".join(map(str, sorted(ds, reverse=True)))); lo = int("".join(map(str, ds)))
    t = R.randint(3, 9); n2 = R.randint(11, 30)
    return ("Digit puzzles.", [
        (f"Use each of the digits {', '.join(map(str, ds))} once to make the largest possible 4-digit number.", str(hi), f"Put the digits in order, largest first: {hi}."),
        (f"Use each of the digits {', '.join(map(str, ds))} once. What is the difference between the largest and the smallest 4-digit numbers you can make?", str(hi - lo), f"Largest: {hi}. Smallest: {lo}. {hi} − {lo} = {hi - lo}."),
        (f"How many 2-digit numbers have digits that add up to {t}?", str(sum(1 for n in range(10, 100) if n // 10 + n % 10 == t)),
         f"List them: {', '.join(str(n) for n in range(10, 100) if n // 10 + n % 10 == t)}.")])

GEN = {"fact": fact, "prime": prime, "place": place, "fdp": fdp, "cash": cash, "rates": rates, "unitc": unitc, "puz1": puz1}
NAME = {"fact": "Factors and multiples", "prime": "Prime numbers", "place": "× and ÷ by 10, 100, 1000", "fdp": "Fractions, decimals and percentages",
        "cash": "Money", "rates": "Recipes, maps and rates", "unitc": "Unit cost and best buys", "puz1": "Number puzzles"}
PLAIN = {"Factors and multiples.", "Prime numbers.", "× and ÷ by 10, 100, 1000.", "Fractions, decimals and percentages.", "Digit puzzles."}

def main():
    qp = sys.argv[1]; Q = json.load(open(qp))
    Q = [q for q in Q if not re.match(r"SK-\w+-N\d", q["id"])]      # safe to re-run
    seen = {q["q"] + "|" + q.get("pic", "") for q in Q}
    for sk, gen in GEN.items():
        have = sum(1 for q in Q if q.get("skill") == sk and not str(q.get("paper", "")).startswith("PT"))
        R = random.Random("new-" + sk); n = 0; tries = 0
        while have < 90 and tries < 6000:
            tries += 1; res = gen(R); stem, parts = res[:2]; pic = res[2] if len(res) > 2 else None
            if len(parts) != 3 or any(not p[1] or "-" in p[1] for p in parts): continue
            texts = [p[0] if stem in PLAIN else f"{stem}\n{'abc'[i]}) {p[0]}" for i, p in enumerate(parts)]
            keys = [t + "|" + (pic or "") for t in texts]
            if keys[0] in seen: continue
            seen.update(keys); n += 1
            for i, p in enumerate(parts):
                d = dict(id=f"SK-{sk}-N{n:02d}{'abc'[i]}", q=texts[i], a=p[1], s=p[2], m=1, wp=False, yr=5, skill=sk, set=f"{sk}-N{n:02d}",
                         lv=["Easy", "Hard", "Challenge"][i], k=NAME[sk], t=NAME[sk], tab=NAME[sk], src="Written for this app in the CSSE style")
                if len(p) > 3 and p[3]: d["wrong"] = [w for w in p[3] if w != p[1]][:3]
                if pic: d["pic"] = pic
                Q.append(d)
            have += 3
        print(f"{NAME[sk]:38} +{n} sets, now {have}")
    json.dump(Q, open(qp, "w"), ensure_ascii=False)

if __name__ == "__main__":
    main()
