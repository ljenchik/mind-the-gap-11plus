"""Linked parts that say "that sequence" / "that number" repeat the numbers from part a,
so a child never has to remember them from the previous screen. Safe to run more than once."""
import json, re, sys
P = sys.argv[1] if len(sys.argv) > 1 else 'questions.json'
qs = json.load(open(P)); ids = {q['id']: q for q in qs}; n = 0
for q in qs:
    m = re.match(r'(.*\d)([b-h])$', q['id'])
    if not m or '\n' in q['q']: continue
    a = ids.get(m.group(1) + 'a')
    if not a: continue
    if 'that sequence' in q['q']:
        prev = ids.get(m.group(1) + chr(ord(m.group(2)) - 1), a)   # "that" = the part just before
        am = re.search(r'sequence starts ([\d, ]+?) and goes up by the same amount each time', prev['q'])
        sm = re.search(r':\s+(.+?)(?:,\s*___)+\s*$', prev['q'])
        if am: lead = f"The sequence {am.group(1)}, … goes up by the same amount each time. "
        elif sm: lead = f"The sequence {sm.group(1)}, … continues in the same way. "
        else: continue
        q['q'] = lead + q['q'].replace('that sequence', 'the sequence'); n += 1
    elif 'that number and the smallest 4-digit number made from the same digits' in q['q']:
        dm = re.search(r'digits ([\d, ]+?) once', a['q'])
        if not dm: continue
        q['q'] = f"Use each of the digits {dm.group(1)} once. What is the difference between the largest and the smallest 4-digit numbers you can make?"
        hi = re.search(r'(\d{4})', a['a']).group(1) if re.search(r'\d{4}', a['a']) else ''
        if hi and not q['s'].startswith('Largest'): q['s'] = f"Largest: {hi}. " + q['s']
        n += 1
def ordn(k): k = int(k); return f"{k}{'th' if 10 <= k % 100 <= 20 else {1: 'st', 2: 'nd', 3: 'rd'}.get(k % 10, 'th')}"
for q in qs:
    for f in ('a', 's'):
        v = q.get(f)
        if isinstance(v, str):
            w = re.sub(r'\b(\d+)(st|nd|rd|th)\b', lambda mm: ordn(mm.group(1)), v)
            if w != v: q[f] = w; n += 1
json.dump(qs, open(P, 'w'), ensure_ascii=False)
print('context fixed:', n)
